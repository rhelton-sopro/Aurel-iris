       ```python
       #!/usr/bin/env python3
       """
       LGPD vocabulary audit for vision-service.

       Mirrors `apps/web/scripts/audit-vocabulary.mjs` (Phase 3+).
       Forbidden tokens (LGPD non-negotiable per PROJECT.md):
         - 'diagnóstico' / 'diagnostico' (with or without accent)
         - 'tratamento'
         - 'cura' (word-boundary — avoid false-positive on 'procurar', 'curativo'... wait,
           'curativo' contains 'cura' — keep word-boundary; 'curativo' is unrelated)

       Scope:
         - vision-service/pipeline/**/*.py
         - vision-service/data/**/*.json
         - vision-service/scripts/**/*.py
         - vision-service/tests/fixtures/**/*.{json,md}

       Exit code:
         - 0 if no forbidden token found
         - 1 if any forbidden token found (CI gate)
       """
       import re
       import sys
       from pathlib import Path

       # Word-boundary on `cura` is critical to avoid false-positives on words like
       # `curativo` (Portuguese: bandage) which legitimately contain the substring.
       # `diagnóstico|diagnostico|tratamento` are matched both with and without accent.
       FORBIDDEN = re.compile(
           r'\bdiagn[oó]stico\b|\btratamento\b|\bcura\b',
           re.IGNORECASE,
       )

       EXTENSIONS = {'.py', '.json', '.md'}
       SCAN_DIRS = ['pipeline', 'data', 'scripts', 'tests/fixtures']
       SKIP_PARTS = {'__pycache__', '.git', '.venv', 'node_modules'}

       ROOT = Path(__file__).resolve().parent.parent  # vision-service/

       def scan_one(path: Path) -> list[str]:
           matches: list[str] = []
           try:
               text = path.read_text(encoding='utf-8')
           except (OSError, UnicodeDecodeError):
               return matches
           for i, line in enumerate(text.splitlines(), start=1):
               # Skip the audit script itself's regex line — would self-trigger.
               if path.name == Path(__file__).name and 'FORBIDDEN' in line:
                   continue
               if FORBIDDEN.search(line):
                   matches.append(f"{path.relative_to(ROOT)}:{i}: {line.strip()}")
           return matches


       def audit() -> int:
           all_matches: list[str] = []
           for d in SCAN_DIRS:
               base = ROOT / d
               if not base.exists():
                   continue
               for path in base.rglob('*'):
                   if not path.is_file():
                       continue
                   if path.suffix not in EXTENSIONS:
                       continue
                   if any(part in SKIP_PARTS for part in path.parts):
                       continue
                   all_matches.extend(scan_one(path))

           if all_matches:
               print('VOCAB FAIL — vocabulário proibido encontrado:')
               for m in all_matches:
                   print(m)
               return 1
           print('OK: vocabulário proibido ausente em vision-service/')
           return 0


       if __name__ == '__main__':
           sys.exit(audit())
       ```

       Note: the self-skip for the regex line is what allows the script to scan itself without false-tripping. Tests verify both the positive (forbidden token detected) and negative (catalog passes) paths.
