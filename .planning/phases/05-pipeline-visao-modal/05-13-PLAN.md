       ```typescript
       import { cookies } from 'next/headers'

       /**
        * Marca finalização do fluxo de captura.
        *
        * Fase 5: após validar 6/6 imagens persistidas, dispara o pipeline Modal
        * via internal fetch para `POST /api/readings/[id]/process`. A rota
        * dedicada lida com auth, ownership, signed URLs, status='processing'
        * placeholder e a chamada Modal (D-T1, plan 05-11).
        *
        * Princípio (D-T1): finalize e process são decoupled. Se o trigger
        * falhar, a leitura permanece em `pending` com as imagens salvas; o
        * terapeuta pode usar o botão "Reprocessar" (plan 05-14) para retentar.
        * NUNCA propagamos a falha de trigger como erro de finalize — isso
        * destruiria o trabalho de captura por uma falha de rede transitória.
        */
       export async function finalizeReadingAction(
         readingId: string
       ): Promise<ReadingFormState> {
         const supabase = await createClient()
         const { data: { user }, error: authError } = await supabase.auth.getUser()
         if (!user || authError) {
           redirect('/login')
         }

         const parsed = readingIdSchema.safeParse({ reading_id: readingId })
         if (!parsed.success) {
           return { error: 'reading_id inválido' }
         }

         // D-T2: dispara trigger via internal fetch, forwarding session cookie
         // para que o auth gate de /api/readings/[id]/process aceite a request.
         const cookieStore = await cookies()
         const cookieHeader = cookieStore.toString()
         const baseUrl =
           process.env.NEXT_PUBLIC_SITE_URL ?? 'http://localhost:3000'
         const triggerUrl = `${baseUrl}/api/readings/${parsed.data.reading_id}/process`

         let triggerStatus: number | null = null
         let triggerError: string | null = null
         try {
           const res = await fetch(triggerUrl, {
             method: 'POST',
             headers: { Cookie: cookieHeader },
             cache: 'no-store',
           })
           triggerStatus = res.status
           if (res.status !== 202) {
             // Capture body for log; do not surface to UI (may contain internal detail).
             const detail = await res.text().catch(() => '')
             console.error(
               `[finalize] trigger non-202 reading=${parsed.data.reading_id} status=${res.status} body=${detail.slice(0, 200)}`,
             )
           }
         } catch (err) {
           triggerError = err instanceof Error ? err.message : 'unknown'
           console.error(
             `[finalize] trigger fetch threw reading=${parsed.data.reading_id}: ${triggerError}`,
           )
         }

         revalidatePath('/leituras')
         revalidatePath(`/leituras/${parsed.data.reading_id}`)

         if (triggerStatus === 202) {
           // Happy path — D-T2: redirect imediato para /leituras; badge "Processando"
           // aparece via 05-14 ao re-renderizar a listagem.
           redirect('/leituras')
         }

         // Soft-warn: imagens salvas, mas trigger não disparou. Reprocessar (05-14) recupera.
         return {
           warning:
             'Captura salva, mas o processamento automático falhou — use Reprocessar na lista para tentar de novo.',
         }
       }
       ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Server Action | Authenticated terapeuta session via Supabase cookies. |
| Server Action → trigger route | Server-to-server `fetch` with explicit cookie forwarding (no implicit credential propagation in Next.js 15 App Router server contexts). |
| Trigger route ownership check | The forwarded session cookie is what makes 05-11's `auth.getUser` + `eq('therapist_id', user.id)` ownership guard work. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-13-01 | Spoofing | Server-side fetch sends a request that bypasses the trigger route's auth gate | mitigate | Cookie forwarding via `next/headers.cookies()` ensures the trigger sees the SAME session as the action; no special internal token is introduced. |
| T-05-13-02 | Privilege escalation | Trigger called for a reading the action's caller does not own | mitigate | The trigger route (05-11) double-checks ownership via `eq('therapist_id', user.id)`. The forwarded cookie carries the action's user identity; spoofing the session cookie would already require breaking Supabase auth. |
| T-05-13-03 | Information disclosure | Trigger error body leaks internal detail to UI | mitigate | Error body is `console.error`'d (server logs only) and truncated to 200 chars; UI only sees the static soft-warn copy. |
| T-05-13-04 | Tampering | finalizeReadingAction propagates trigger failure as a hard error, destroying captura | mitigate | All non-202 paths and all fetch-throws are caught and returned as `warning`; redirect happens ONLY on 202. Captura is preserved (D-T1 decoupling). |
| T-05-13-05 | Denial of service | Hot-loop firing trigger on every finalize even if pipeline is queued | accept | finalize is user-initiated (one click after 6/6 captures). Trigger route has its own status guard (only `pending`/`failed` accepted by 05-11). Repeated finalize on already-`processing` reading would 404 from the route, surfacing as soft-warn (acceptable). |
| T-05-13-06 | Repudiation | Cannot tell which finalize call drove which Modal run | mitigate | Server logs `[finalize] trigger ... reading=<id> status=<n>` correlate with `[process] ... reading=<id>` (05-11 logs) and `[webhook] applied reading=<id> call_id=<id>` (05-12 logs). |
</threat_model>
