       ```python
       """U-Net iris segmentation (Phase 07.1.5 Approach C fallback per D-02)."""
       from __future__ import annotations
       from pathlib import Path
       import os
       import cv2
       import numpy as np

       _unet_model = None
       _DEFAULT_UNET_PATH = "/models/iris_unet.pth"


       def _resolve_model_path() -> str:
           env = os.environ.get("IRIS_UNET_PATH")
           if env and Path(env).is_file():
               return env
           return _DEFAULT_UNET_PATH


       def get_unet():
           """Lazy singleton — load smp.Unet only on first call."""
           global _unet_model
           if _unet_model is None:
               import torch
               import segmentation_models_pytorch as smp
               model = smp.Unet(
                   encoder_name="mobilenet_v3_small",
                   encoder_weights=None,
                   classes=1,
                   in_channels=3,
               )
               state_dict = torch.load(
                   _resolve_model_path(),
                   map_location="cpu",
                   weights_only=True,  # RESEARCH Security Domain — never set False
               )
               model.load_state_dict(state_dict)
               model.eval()
               _unet_model = model
           return _unet_model


       def predict_iris_mask(rgb: np.ndarray) -> np.ndarray:
           """Predict iris binary mask from RGB input.

           Args:
               rgb: H x W x 3 uint8 RGB.

           Returns:
               uint8 H x W mask (0 or 255), scaled back to input resolution.
           """
           if rgb.dtype != np.uint8 or rgb.ndim != 3 or rgb.shape[2] != 3:
               raise ValueError(
                   f"predict_iris_mask: expected uint8 HxWx3, got {rgb.dtype} {rgb.shape}"
               )
           import torch
           model = get_unet()
           h, w = rgb.shape[:2]
           # Resize to model input size (per addendum — typically 256x256 for MobileNetV3-small)
           target = 256
           small = cv2.resize(rgb, (target, target), interpolation=cv2.INTER_AREA)
           # Normalize per ImageNet
           x = small.astype(np.float32) / 255.0
           mean = np.array([0.485, 0.456, 0.406], dtype=np.float32)
           std = np.array([0.229, 0.224, 0.225], dtype=np.float32)
           x = (x - mean) / std
           x = x.transpose(2, 0, 1)[None, ...]  # (1, 3, H, W)
           with torch.no_grad():
               logits = model(torch.from_numpy(x))
               probs = torch.sigmoid(logits)[0, 0].cpu().numpy()
           mask_small = (probs >= 0.5).astype(np.uint8) * 255
           mask = cv2.resize(mask_small, (w, h), interpolation=cv2.INTER_NEAREST)
           return mask


       def predict_iris_circle(rgb: np.ndarray) -> dict | None:
           """Fit a minimum-enclosing circle on the U-Net mask.

           Returns:
               {"center": (cx, cy), "radius": r} or None if mask is empty.
           """
           mask = predict_iris_mask(rgb)
           contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
           if not contours:
               return None
           largest = max(contours, key=cv2.contourArea)
           (cx, cy), r = cv2.minEnclosingCircle(largest)
           if r < 1.0:
               return None
           return {"center": (float(cx), float(cy)), "radius": float(r)}
       ```
