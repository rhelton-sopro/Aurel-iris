       ```sql
       -- Fase 11.1 (2026-05-26): therapist_invites + clients UNIQUE
       --
       -- D-DUPE: bloqueia convite a email já cadastrado em auth.users (check via inviteTherapistAction, não DB constraint cross-schema)
       -- D-TTL: token expira em 7 dias (expires_at = created_at + interval '7 days', validado em /convite-terapeuta/[token])
       -- D-CLIENT: UNIQUE(therapist_id, email) per terapeuta (não global); resolve memory tech debt project_clients_unique_email_tech_debt

       -- 1. Tabela therapist_invites
       create table if not exists public.therapist_invites (
         token uuid primary key default gen_random_uuid(),
         email citext not null,
         invited_by uuid not null references public.profiles(id) on delete cascade,
         created_at timestamptz not null default now(),
         expires_at timestamptz not null default (now() + interval '7 days'),
         used_at timestamptz null,
         used_by_user_id uuid null references auth.users(id) on delete set null
       );

       -- 2. Indexes
       create index if not exists therapist_invites_email_idx on public.therapist_invites (email);
       create index if not exists therapist_invites_token_active_idx
         on public.therapist_invites (token, expires_at, used_at)
         where used_at is null;
       create index if not exists therapist_invites_invited_by_idx on public.therapist_invites (invited_by);

       -- 3. RLS habilitada — founder-only via service role na admin server action.
       --    Client roles NÃO podem ler/escrever. RPC /convite-terapeuta/[token] usa service client.
       alter table public.therapist_invites enable row level security;

       -- Sem policies pra authenticated/anon — bloqueio total. Service role bypass naturalmente.
       -- Defense-in-depth: garante que mesmo se o anon key vazar, ninguém liste tokens.

       -- 4. COMMENT documentando intent
       comment on table public.therapist_invites is
         'Fase 11.1: tokens de convite para terapeutas via /convite-terapeuta/[token]. Founder gera via /admin/terapeutas (inviteTherapistAction); terapeuta abre URL + completa form + verifica OTP. Token marca used_at quando submit OK. Expira em 7d (D-TTL). RLS bloqueia tudo exceto service role.';

       comment on column public.therapist_invites.token is
         'uuid PK gerado pelo DB (gen_random_uuid). Forma da URL: /convite-terapeuta/{token}.';

       comment on column public.therapist_invites.email is
         'citext (case-insensitive) — email do terapeuta convidado. Pré-checado em auth.users por inviteTherapistAction (D-DUPE: bloqueia se já existe).';

       comment on column public.therapist_invites.invited_by is
         'profiles.id do founder que gerou o convite. Cascade ON DELETE — se founder deletado, convites somem.';

       comment on column public.therapist_invites.expires_at is
         'D-TTL: created_at + 7 dias (default). GET na rota /convite-terapeuta/[token] verifica expires_at > now() ou retorna 404.';

       comment on column public.therapist_invites.used_at is
         'Timestamp do markTherapistInviteUsedAction (pós-verifyOtp). NULL = token ainda usável. Non-NULL = idempotência (token 1× uso).';

       comment on column public.therapist_invites.used_by_user_id is
         'auth.users.id do user que ativou o convite. SET NULL se user deletado (cascade pra preservar histórico).';

       -- 5. UNIQUE constraint em clients (D-CLIENT)
       alter table public.clients
         add constraint clients_therapist_email_unique unique (therapist_id, email);

       comment on constraint clients_therapist_email_unique on public.clients is
         'Fase 11.1 D-CLIENT (2026-05-26): per terapeuta, email único. NÃO global — terapeutas diferentes podem ter mesmo cliente. createClientAction trata violation com mensagem amigável.';
       ```
