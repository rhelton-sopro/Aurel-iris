```typescript
'use client'

import * as React from 'react'
import Link from 'next/link'
import { useTransition, useState } from 'react'
import { Clock, Loader2, X } from 'lucide-react'
import { formatDistanceToNow } from 'date-fns'
import { ptBR } from 'date-fns/locale'
import { Button, buttonVariants } from '@/components/ui/button'
import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert'
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog'
import { discardReadingAction } from '@/app/actions/readings'
import { cn } from '@/lib/utils'

interface RecoveryBannerProps {
  draft: {
    id: string
    created_at: string
    client_name: string
    imagesCaptured: number
  }
}

export function RecoveryBanner({ draft }: RecoveryBannerProps) {
  const [dismissed, setDismissed] = React.useState(false)
  const [confirmOpen, setConfirmOpen] = React.useState(false)
  const [discardError, setDiscardError] = useState<string | null>(null)
  const [isPending, startTransition] = useTransition()

  if (dismissed) return null

  const startedAgo = formatDistanceToNow(new Date(draft.created_at), {
    addSuffix: true,
    locale: ptBR,
  })

  function handleDiscard() {
    setDiscardError(null)
    startTransition(async () => {
      const result = await discardReadingAction(draft.id)
      if (result?.error) {
        setDiscardError(result.error)
        return
      }
      setConfirmOpen(false)
      setDismissed(true)
    })
  }

  return (
    <>
      <Alert className="mb-6 bg-amber-50 border-amber-200">
        <Clock className="h-4 w-4 text-amber-700" />
        <button
          type="button"
          onClick={() => setDismissed(true)}
          aria-label="Dispensar aviso"
          className="absolute right-2 top-2 p-1 text-amber-700 hover:bg-amber-100 rounded"
        >
          <X className="h-4 w-4" />
        </button>
        <AlertTitle className="text-amber-900">Você tem uma leitura em andamento</AlertTitle>
        <AlertDescription className="text-amber-900">
          <p>
            Leitura de <strong>{draft.client_name}</strong> iniciada {startedAgo} — {draft.imagesCaptured} de 6 imagens registradas. Deseja continuar?
          </p>
          <div className="mt-3 flex flex-wrap gap-2">
            <Link
              href={`/leituras/nova/capturar?reading=${draft.id}&resume=true`}
              className={cn(buttonVariants({ size: 'sm' }))}
            >
              Continuar leitura
            </Link>
            <Button variant="ghost" size="sm" onClick={() => setConfirmOpen(true)}>
              Descartar
            </Button>
          </div>
        </AlertDescription>
      </Alert>

      <Dialog open={confirmOpen} onOpenChange={(open) => { if (!isPending) setConfirmOpen(open) }}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>Descartar leitura em andamento?</DialogTitle>
            <DialogDescription>
              As <strong>{draft.imagesCaptured} imagens já registradas</strong> serão apagadas permanentemente. Esta ação não pode ser desfeita.
            </DialogDescription>
          </DialogHeader>
          {discardError && (
            <p className="text-sm text-destructive">{discardError}</p>
          )}
          <DialogFooter>
            <Button variant="outline" onClick={() => setConfirmOpen(false)} disabled={isPending}>
              Cancelar
            </Button>
            <Button
              variant="destructive"
              onClick={handleDiscard}
              disabled={isPending}
              aria-busy={isPending}
            >
              {isPending ? (
                <>
                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                  Descartando...
                </>
              ) : (
                'Sim, descartar'
              )}
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </>
  )
}
```

import Link from 'next/link'
import { format } from 'date-fns'
import { ptBR } from 'date-fns/locale'
import { ChevronRight, FileText } from 'lucide-react'
import { createClient } from '@/lib/supabase/server'
import { Badge } from '@/components/ui/badge'
import { buttonVariants } from '@/components/ui/button'
import { cn } from '@/lib/utils'

export default async function LeiturasPage() {
  const supabase = await createClient()
  // RLS filtra para o terapeuta logado
  const { data: readings } = await supabase
    .from('readings')
    .select('id, status, created_at, client_id, clients(full_name), reading_images(count)')
    .order('created_at', { ascending: false })
    .limit(50)

  const list = readings ?? []

  return (
    <div className="space-y-6 max-w-3xl">
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-semibold">Leituras</h1>
        <Link href="/leituras/nova" className={cn(buttonVariants())}>
          Nova leitura
        </Link>
      </div>

      {list.length === 0 ? (
        <div className="flex flex-col items-center justify-center py-16 gap-3 text-center">
          <FileText className="h-12 w-12 text-muted-foreground" />
          <p className="text-lg font-medium">Nenhuma leitura ainda</p>
          <p className="text-sm text-muted-foreground">
            Crie a primeira leitura tocando em &ldquo;Nova leitura&rdquo;.
          </p>
        </div>
      ) : (
        <div className="border rounded-lg divide-y">
          {list.map(r => {
            const clientObj = Array.isArray(r.clients) ? r.clients[0] : r.clients
            const clientName = clientObj?.full_name ?? 'Cliente'
            const imagesCount = (r.reading_images?.[0]?.count as number | undefined) ?? 0
            const isDraft = r.status === 'pending' && imagesCount < 6
            const href = isDraft
              ? `/leituras/nova/capturar?reading=${r.id}&resume=true`
              : `/leituras` // detalhe vem em fase futura
            return (
              <Link
                key={r.id}
                href={href}
                className="flex items-center justify-between gap-3 px-4 py-3 hover:bg-muted/50 transition-colors"
              >
                <div className="flex flex-col gap-1 min-w-0">
                  <span className="text-sm font-medium truncate">{clientName}</span>
                  <span className="text-xs text-muted-foreground">
                    {format(new Date(r.created_at), "d 'de' MMM 'de' yyyy, HH:mm", { locale: ptBR })}
                    {' · '}
                    {imagesCount} de 6 imagens
                  </span>
                </div>
                <div className="flex items-center gap-2 shrink-0">
                  {isDraft ? (
                    <Badge variant="outline" className="bg-amber-50 text-amber-900 border-amber-200">
                      Em andamento
                    </Badge>
                  ) : (
                    <Badge variant="outline">{statusLabel(r.status)}</Badge>
                  )}
                  <ChevronRight className="h-4 w-4 text-muted-foreground" />
                </div>
              </Link>
            )
          })}
        </div>
      )}
    </div>
  )
}

function statusLabel(status: string | null): string {
  switch (status) {
    case 'pending': return 'Pendente'
    case 'processing': return 'Processando'
    case 'ready': return 'Pronta'
    case 'failed': return 'Falhou'
    case 'edited': return 'Editada'
    default: return 'Desconhecido'
  }
}

const STORAGE_KEY = 'pwa_install_banner_dismissed_at'
const DISMISS_TTL_MS = 7 * 24 * 60 * 60 * 1000 // 7 dias (D-14)

function isRecentlyDismissed(): boolean {
  if (typeof window === 'undefined') return true
  const raw = window.localStorage.getItem(STORAGE_KEY)
  if (!raw) return false
  const ts = Number(raw)
  if (!Number.isFinite(ts)) return false
  return Date.now() - ts < DISMISS_TTL_MS
}

function markDismissed(): void {
  if (typeof window === 'undefined') return
  window.localStorage.setItem(STORAGE_KEY, String(Date.now()))
}

export function PWAInstallBanner() {
  const { isStandalone, isIOS, canPromptAndroid, promptInstall } = usePWAInstall()
  const [dismissed, setDismissed] = React.useState(true) // default true → não pisca antes de hydrate

  React.useEffect(() => {
    setDismissed(isRecentlyDismissed())
  }, [])

  function handleDismiss() {
    markDismissed()
    setDismissed(true)
  }

  async function handleAndroidInstall() {
    const accepted = await promptInstall()
    if (accepted) {
      // Não precisa marcar dismiss; isStandalone passará a true após appinstalled
      setDismissed(true)
    }
  }

  // Não renderizar se: já instalado, dismissado nos últimos 7 dias, ou nem Android com prompt nem iOS
  if (isStandalone) return null
  if (dismissed) return null
  if (!isIOS && !canPromptAndroid) return null

  return (
    <Alert className="bg-card border shadow-md relative">
      <button
        type="button"
        onClick={handleDismiss}
        aria-label="Dispensar"
        className="absolute right-2 top-2 p-1 text-muted-foreground hover:text-foreground rounded"
      >
        <X className="h-4 w-4" />
      </button>
      <AlertTitle>Instale o Aurel Iris no celular</AlertTitle>
      <AlertDescription className="space-y-3">
        <p className="text-sm">
          A captura funciona melhor com o app instalado na tela inicial. Tela cheia, sem barras do navegador.
        </p>
        <div className="flex flex-wrap gap-2">
          {isIOS ? (
            <Sheet>
              <SheetTrigger asChild>
                <Button size="sm">Como instalar</Button>
              </SheetTrigger>
              <SheetContent side="bottom" className="space-y-4">
                <SheetHeader>
                  <SheetTitle>Instalar Aurel Iris no iPhone</SheetTitle>
                  <SheetDescription>
                    Para instalar: toque em <Share className="inline h-4 w-4 align-middle" /> (compartilhar) na barra do Safari, role e selecione <strong>&ldquo;Adicionar à Tela de Início&rdquo;</strong>.
                  </SheetDescription>
                </SheetHeader>
                <div className="flex justify-center pt-4">
                  <Share className="h-12 w-12 text-foreground" aria-hidden="true" />
                </div>
              </SheetContent>
            </Sheet>
          ) : canPromptAndroid ? (
            <Button size="sm" onClick={() => void handleAndroidInstall()}>
              Instalar app
            </Button>
          ) : null}
          <Button size="sm" variant="ghost" onClick={handleDismiss}>
            Agora não
          </Button>
        </div>
      </AlertDescription>
    </Alert>
  )
}
