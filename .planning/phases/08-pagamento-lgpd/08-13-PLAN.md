```typescript
import 'server-only'
import { createServiceClient } from '@/lib/supabase/service'
import { logAuditEvent } from '@/lib/audit/log'
import { notifyCreditExpiring, type ExpiryWindow } from '@/lib/notifications/notify-credit-expiring'

/**
 * Cron job — libera reservations expiradas (status='active' + expires_at < now).
 * Chama RPC release_reservation pra cada uma (atomic: devolve saldo + status='expired').
 */
export async function releaseExpiredReservations(): Promise<{ released: number; errors: number }> {
  const service = createServiceClient()
  const { data: expired } = await service
    .from('credit_reservations')
    .select('reading_id')
    .eq('status', 'active')
    .lt('expires_at', new Date().toISOString())
    .limit(500)  // safety cap

  let released = 0, errors = 0
  for (const r of expired ?? []) {
    const { error } = await service.rpc('release_reservation', {
      p_reading_id: r.reading_id, p_reason: 'expired',
    })
    if (error) { errors++; continue }
    released++
  }
  return { released, errors }
}

/**
 * Cron job — marca credits 12m+ como expired + zera saldo.
 * Idempotent: WHERE status='active' AND expires_at < now.
 */
export async function expireOldCredits(): Promise<{ expired: number }> {
  const service = createServiceClient()
  const { data, error } = await service
    .from('customer_credits')
    .update({ status: 'expired', leituras_remaining: 0 })
    .eq('status', 'active')
    .lt('expires_at', new Date().toISOString())
    .select('id, user_id, leituras_remaining')
  if (error) {
    console.error('[cron] expireOldCredits failed:', error.message)
    return { expired: 0 }
  }
  for (const row of data ?? []) {
    await service.from('credit_transactions').insert({
      user_id: row.user_id, credit_id: row.id, type: 'expire',
      amount: -row.leituras_remaining,
      notes: 'cron daily — 12m validade',
    })
    await logAuditEvent({
      event_type: 'credit.expired',
      actor_user_id: null, actor_email: 'system',
      target_type: 'credit', target_id: row.id,
      metadata: { leituras_lost: row.leituras_remaining },
    })
  }
  return { expired: data?.length ?? 0 }
}

/**
 * Cron job — marca trials 60d+ como ended (first-wins via days).
 * Reservations criadas durante trial e ainda active passam por releaseExpiredReservations.
 */
export async function expireOldTrials(): Promise<{ ended: number }> {
  const service = createServiceClient()
  const { data, error } = await service
    .from('trial_status')
    .update({ ended_at: new Date().toISOString(), ended_reason: 'days_elapsed' })
    .is('ended_at', null)
    .lt('trial_expires_at', new Date().toISOString())
    .select('user_id')
  if (error) {
    console.error('[cron] expireOldTrials failed:', error.message)
    return { ended: 0 }
  }
  for (const row of data ?? []) {
    await logAuditEvent({
      event_type: 'trial.ended',
      actor_user_id: row.user_id, target_type: 'profile', target_id: row.user_id,
      metadata: { reason: 'days_elapsed' },
    })
  }
  return { ended: data?.length ?? 0 }
}

/**
 * Cron job — envia avisos de expiração 30d/7d/0d.
 *
 * Dedup via audit_events com event_type='credit.expiring_warning' (NÃO reuse
 * 'credit.expired' — WARN-4: 'credit.expired' marca o evento TERMINAL de credit;
 * warnings são um sub-tipo separado pra não distorcer analytics).
 *
 * audit_events.metadata.notification_days distingue qual window (30/7/0).
 */
const WINDOWS: { days: ExpiryWindow; start: number; end: number }[] = [
  { days: 30, start: 30, end: 31 },  // exatamente 30..31 dias remaining
  { days: 7, start: 7, end: 8 },
  { days: 0, start: 0, end: 1 },  // hoje
]

export async function sendExpirationWarnings(): Promise<{ sent: number; skipped: number }> {
  const service = createServiceClient()
  const now = Date.now()
  let sent = 0, skipped = 0

  for (const win of WINDOWS) {
    const lower = new Date(now + win.start * 86400000).toISOString()
    const upper = new Date(now + win.end * 86400000).toISOString()

    const { data: credits } = await service
      .from('customer_credits')
      .select('id, user_id, expires_at, leituras_remaining, credit_packages(name), profiles!inner(email, full_name)')
      .eq('status', 'active')
      .gte('expires_at', lower)
      .lt('expires_at', upper)
      .gt('leituras_remaining', 0)

    for (const c of credits ?? []) {
      // Dedup: check audit_events pra warning prévio com mesmo credit_id + same window.
      // event_type='credit.expiring_warning' garante que NÃO colide com row terminal 'credit.expired'.
      const { data: existing } = await service
        .from('audit_events')
        .select('id')
        .eq('event_type', 'credit.expiring_warning')
        .eq('target_id', c.id)
        .contains('metadata', { notification_days: win.days })
        .maybeSingle()
      if (existing) { skipped++; continue }

      const enriched = c as unknown as {
        id: string; user_id: string; expires_at: string; leituras_remaining: number;
        credit_packages: { name: string };
        profiles: { email: string; full_name: string | null };
      }
      void notifyCreditExpiring({
        userEmail: enriched.profiles.email,
        userName: enriched.profiles.full_name,
        packageName: enriched.credit_packages.name,
        leituras_remaining: enriched.leituras_remaining,
        expiresAt: enriched.expires_at,
        daysOut: win.days,
      }).catch(() => {})

      await logAuditEvent({
        event_type: 'credit.expiring_warning',  // ← distinto de 'credit.expired'
        actor_user_id: enriched.user_id, actor_email: 'system',
        target_type: 'credit', target_id: enriched.id,
        metadata: { notification_days: win.days, expires_at: enriched.expires_at },
      })
      sent++
    }
  }
  return { sent, skipped }
}
```

```typescript
import { NextResponse, type NextRequest } from 'next/server'
import {
  releaseExpiredReservations,
  expireOldCredits,
  expireOldTrials,
  sendExpirationWarnings,
} from '@/lib/billing/cron-jobs'

export const runtime = 'nodejs'
export const maxDuration = 60  // way under 800s

export async function GET(request: NextRequest): Promise<NextResponse> {
  // Vercel Cron auto-injects Authorization: Bearer ${CRON_SECRET}
  const auth = request.headers.get('authorization')
  if (auth !== `Bearer ${process.env.CRON_SECRET}`) {
    console.warn('[cron-daily] AUTH_REJECTED')
    return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
  }

  // Sequential — each is idempotent + bounded (limit 500)
  const results = {
    reservations: await releaseExpiredReservations().catch(err => ({ released: 0, errors: 1, error: String(err) })),
    credits: await expireOldCredits().catch(err => ({ expired: 0, error: String(err) })),
    trials: await expireOldTrials().catch(err => ({ ended: 0, error: String(err) })),
    warnings: await sendExpirationWarnings().catch(err => ({ sent: 0, skipped: 0, error: String(err) })),
  }
  console.info(`[cron-daily] BATCH_COMPLETE ${JSON.stringify(results)}`)
  return NextResponse.json({ ok: true, ...results })
}
```
