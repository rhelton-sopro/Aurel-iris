```typescript
import 'server-only'
import { createServiceClient } from '@/lib/supabase/service'

export type TermoGateResult =
  | { ok: true; signed_at: string; document_url: string | null }
  | { ok: false; reason: 'termo_missing' | 'client_not_found' | 'db_error'; detail?: string }

/**
 * BILLING-03 + LGPD-01 (D-19) — gate que bloqueia captura biométrica sem termo assinado.
 *
 * Source of truth: `clients.consent_signed_at`. Setado por signTermAction (08-08) quando
 * cliente assina o termo via /convite OR terapeuta assina em office_handoff.
 *
 * NÃO consulta client_consents direto — esse table é append-only e pode ter rows revoke/etc.
 * `clients.consent_signed_at` é o snapshot operacional.
 *
 * Caller convention: SE { ok: false, reason: 'termo_missing' }, UI deve oferecer
 * link/CTA pra TermoBiometricoStep (componente 08-08) com o clientId.
 */
export async function assertClientTermoSigned(clientId: string): Promise<TermoGateResult> {
  const service = createServiceClient()
  const { data, error } = await service
    .from('clients')
    .select('consent_signed_at, consent_document_url')
    .eq('id', clientId)
    .maybeSingle()

  if (error) {
    console.error(`[termo-gate] db error client=${clientId}:`, error.message)
    return { ok: false, reason: 'db_error', detail: error.message }
  }
  if (!data) {
    return { ok: false, reason: 'client_not_found' }
  }
  if (!data.consent_signed_at) {
    return { ok: false, reason: 'termo_missing' }
  }
  return {
    ok: true,
    signed_at: data.consent_signed_at,
    document_url: data.consent_document_url ?? null,
  }
}
```
