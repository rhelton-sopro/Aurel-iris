  <behavior>
    - computeRefundValue({ purchase_date, price_brl, leituras_purchased, leituras_remaining, leituras_reserved }, now) →
      - se now > purchase_date + 7d → { eligible: false, reason: 'window_expired', value_brl: 0 }
      - se leituras_consumed == 0 (purchased == remaining+reserved) → { eligible: true, kind: 'total', value_brl: price_brl }
      - se 0 < leituras_consumed < purchased → proporcional: unit_price * (remaining+reserved), kind: 'partial'
      - se leituras_consumed == purchased → { eligible: false, reason: 'no_balance', value_brl: 0 }
    - Edge: precisão 2 decimais (round half-even ou simple Math.round 2dp)
  </behavior>

**Arquivo principal:**

```typescript
import { refundWindowEndsAt } from './config'

export type RefundPolicy =
  | { eligible: true; kind: 'total'; value_brl: number; leituras_to_refund: number }
  | { eligible: true; kind: 'partial'; value_brl: number; leituras_to_refund: number; unit_price_brl: number }
  | { eligible: false; reason: 'window_expired' | 'no_balance' | 'wrong_status'; value_brl: 0 }

export interface CreditForRefund {
  purchase_date: string  // ISO timestamp
  price_brl: number  // valor total pago (do package)
  leituras_purchased: number
  leituras_remaining: number
  leituras_reserved: number
  status: string  // 'active' | 'expired' | 'refunded' | 'pending'
}

/**
 * Calcula refund value per D-13:
 *   - 0 consumidas → refund total
 *   - 1..N-1 consumidas → proporcional: unit_price × (remaining + reserved)
 *   - todas consumidas → no_balance
 *   - fora da janela 7d → window_expired
 */
export function computeRefundValue(
  credit: CreditForRefund,
  now: Date = new Date(),
): RefundPolicy {
  if (credit.status !== 'active') {
    return { eligible: false, reason: 'wrong_status', value_brl: 0 }
  }

  const purchasedAt = new Date(credit.purchase_date)
  const windowEnd = refundWindowEndsAt(purchasedAt)
  if (now > windowEnd) {
    return { eligible: false, reason: 'window_expired', value_brl: 0 }
  }

  const consumed = credit.leituras_purchased - credit.leituras_remaining - credit.leituras_reserved
  if (consumed >= credit.leituras_purchased) {
    return { eligible: false, reason: 'no_balance', value_brl: 0 }
  }
  if (consumed === 0) {
    return {
      eligible: true,
      kind: 'total',
      value_brl: credit.price_brl,
      leituras_to_refund: credit.leituras_purchased,
    }
  }
  // partial
  const unit_price = credit.price_brl / credit.leituras_purchased
  const refundable = credit.leituras_remaining + credit.leituras_reserved
  const value = Math.round(unit_price * refundable * 100) / 100  // 2dp
  return {
    eligible: true,
    kind: 'partial',
    value_brl: value,
    leituras_to_refund: refundable,
    unit_price_brl: Math.round(unit_price * 100) / 100,
  }
}
```

import { describe, expect, it } from 'vitest'

const now = new Date('2026-05-27T12:00:00Z')

describe('computeRefundValue', () => {
  it('total refund when 0 consumed within 7d window', () => {
    const r = computeRefundValue({
      purchase_date: '2026-05-25T00:00:00Z',  // 2d ago
      price_brl: 298.50,
      leituras_purchased: 5, leituras_remaining: 5, leituras_reserved: 0,
      status: 'active',
    }, now)
    expect(r).toEqual({ eligible: true, kind: 'total', value_brl: 298.50, leituras_to_refund: 5 })
  })

  it('partial proportional when 2 consumed (5 → 3 remaining)', () => {
    const r = computeRefundValue({
      purchase_date: '2026-05-25T00:00:00Z',
      price_brl: 298.50,
      leituras_purchased: 5, leituras_remaining: 3, leituras_reserved: 0,
      status: 'active',

    expect(r.eligible).toBe(true)
    if (r.eligible && r.kind === 'partial') {
      expect(r.unit_price_brl).toBe(59.70)  // 298.50 / 5
      expect(r.value_brl).toBe(179.10)  // 59.70 × 3
      expect(r.leituras_to_refund).toBe(3)

      purchase_date: '2026-05-25T00:00:00Z',

      leituras_purchased: 5, leituras_remaining: 2, leituras_reserved: 1,
      status: 'active',

      expect(r.leituras_to_refund).toBe(3)  // 2 + 1
      expect(r.value_brl).toBe(179.10)

```typescript
'use server'
import 'server-only'

import { revalidatePath } from 'next/cache'
import { createClient } from '@/lib/supabase/server'
import { createServiceClient } from '@/lib/supabase/service'
import { createAsaasCustomer, createAsaasPayment, refundAsaasPayment } from '@/lib/asaas/client'
import { computeRefundValue } from '@/lib/billing/refund-policy'
import { logAuditEvent } from '@/lib/audit/log'
import { cpfDigits } from '@/lib/auth/cpf'
import {
  createChargeSchema, refundPackageSchema,
  type CreateChargeInput, type CreateChargeResult,
  type RefundPackageInput, type RefundPackageResult,
} from './billing.schemas'

export async function createChargeAction(input: CreateChargeInput): Promise<CreateChargeResult> {
  // 1. Session gate
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { ok: false, error: 'Não autenticado.' }

  // 2. Zod
  const parsed = createChargeSchema.safeParse(input)
  if (!parsed.success) return { ok: false, error: 'SKU inválido.' }

  // 3. Load package + profile in parallel
  const [pkgRes, profileRes] = await Promise.all([
    supabase.from('credit_packages').select('id, sku, name, leituras_count, price_brl').eq('sku', parsed.data.sku).eq('active', true).maybeSingle(),
    supabase.from('profiles').select('id, asaas_customer_id, full_name, cpf, phone, email').eq('id', user.id).maybeSingle(),
  ])
  if (!pkgRes.data) return { ok: false, error: 'Pacote não encontrado.' }
  if (!profileRes.data) return { ok: false, error: 'Perfil não encontrado.' }
  const pkg = pkgRes.data
  const profile = profileRes.data
  if (!profile.cpf || !profile.phone) {
    return { ok: false, error: 'Complete CPF + telefone no seu perfil antes de comprar.' }
  }

  // 4. Asaas customer — create if missing
  const service = createServiceClient()
  let asaasCustomerId = profile.asaas_customer_id
  if (!asaasCustomerId) {
    const cust = await createAsaasCustomer({
      name: profile.full_name ?? user.email ?? 'Terapeuta',
      cpfCnpj: cpfDigits(profile.cpf),
      email: user.email ?? '',
      mobilePhone: cpfDigits(profile.phone),  // só dígitos
      externalReference: profile.id,
    })
    if (!cust.ok) {
      console.error('[billing] createAsaasCustomer failed:', cust.error)
      return { ok: false, error: 'Falha ao criar cliente Asaas: ' + cust.error }
    }
    asaasCustomerId = cust.data.id
    await service.from('profiles').update({ asaas_customer_id: asaasCustomerId }).eq('id', profile.id)
  }

  // 5. INSERT customer_credits pending (BEFORE Asaas payment — needed for externalReference)
  const { data: pendingCredit, error: insErr } = await service.from('customer_credits').insert({
    user_id: profile.id,
    package_id: pkg.id,
    leituras_purchased: pkg.leituras_count,
    leituras_remaining: 0,  // becomes pkg.leituras_count on webhook PAYMENT_CONFIRMED
    leituras_reserved: 0,
    expires_at: new Date(Date.now() + 365 * 86400000).toISOString(),  // placeholder; webhook resets to confirmed_at+12m
    status: 'pending',
  }).select('id').single()
  if (insErr || !pendingCredit) {
    return { ok: false, error: 'Erro interno ao registrar compra.' }
  }

  // 6. Create Asaas payment
  const dueDate = new Date(Date.now() + 7 * 86400000).toISOString().slice(0, 10)
  const payment = await createAsaasPayment({
    customer: asaasCustomerId,
    billingType: 'UNDEFINED',  // cliente escolhe no checkout
    value: pkg.price_brl,
    dueDate,
    description: `Iris Codex — ${pkg.name} (${pkg.leituras_count} ${pkg.leituras_count === 1 ? 'leitura' : 'leituras'})`,
    externalReference: pendingCredit.id,
  })
  if (!payment.ok) {
    // Compensate: delete pending credit (Asaas didn't accept)
    await service.from('customer_credits').delete().eq('id', pendingCredit.id)
    return { ok: false, error: 'Falha ao criar cobrança: ' + payment.error }
  }

  // 7. Update credit with asaas_payment_id (unique key per RESEARCH pitfall #5)
  await service.from('customer_credits').update({
    asaas_payment_id: payment.data.id,
    asaas_invoice_url: payment.data.invoiceUrl,
    asaas_payment_status: payment.data.status,
  }).eq('id', pendingCredit.id).eq('user_id', profile.id)  // defensive

  await logAuditEvent({
    event_type: 'credit.purchase_initiated',
    actor_user_id: profile.id, actor_email: user.email,
    target_type: 'credit', target_id: pendingCredit.id,
    metadata: { sku: pkg.sku, asaas_payment_id: payment.data.id, value_brl: pkg.price_brl },
  })

  console.info(`[billing] CHARGE_CREATED user=${profile.id} sku=${pkg.sku} payment=${payment.data.id}`)
  revalidatePath('/assinatura')
  return {
    ok: true,
    credit_id: pendingCredit.id,
    invoice_url: payment.data.invoiceUrl ?? '',
    asaas_payment_id: payment.data.id,
  }
}

export async function refundPackageAction(input: RefundPackageInput): Promise<RefundPackageResult> {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return { ok: false, error: 'Não autenticado.' }

  const parsed = refundPackageSchema.safeParse(input)
  if (!parsed.success) return { ok: false, error: 'Input inválido.' }

  // 1. SELECT credit via session client (RLS enforces user_id = auth.uid())
  const { data: credit, error: selErr } = await supabase
    .from('customer_credits')
    .select('id, user_id, asaas_payment_id, purchase_date, leituras_purchased, leituras_remaining, leituras_reserved, status, credit_packages(price_brl)')
    .eq('id', parsed.data.credit_id)
    .maybeSingle()
  if (selErr || !credit) return { ok: false, error: 'Crédito não encontrado.' }

  // 2. Compute policy
  const policy = computeRefundValue({
    purchase_date: credit.purchase_date,
    price_brl: (credit as { credit_packages: { price_brl: number } }).credit_packages.price_brl,
    leituras_purchased: credit.leituras_purchased,
    leituras_remaining: credit.leituras_remaining,
    leituras_reserved: credit.leituras_reserved,
    status: credit.status,
  })
  if (!policy.eligible) {
    const msg = policy.reason === 'window_expired'
      ? 'Janela de arrependimento expirada (7 dias).'
      : policy.reason === 'no_balance'
      ? 'Pacote já foi totalmente consumido.'
      : 'Pacote não pode ser reembolsado neste estado.'
    return { ok: false, error: msg }
  }

  // 3. Call Asaas refund — body={} for total, body.value for partial
  if (!credit.asaas_payment_id) return { ok: false, error: 'Pagamento Asaas não associado.' }
  const refundBody = policy.kind === 'total'
    ? undefined
    : { value: policy.value_brl, description: `Iris Codex — arrependimento 7d (${policy.leituras_to_refund} leituras restantes)` }
  const refund = await refundAsaasPayment(credit.asaas_payment_id, refundBody)
  if (!refund.ok) return { ok: false, error: 'Falha no Asaas: ' + refund.error }

  // 4. Update local state — webhook PAYMENT_REFUNDED OR PARTIALLY_REFUNDED também atualizará; mas ser proativo
  const service = createServiceClient()
  if (policy.kind === 'total') {
    await service.from('customer_credits')
      .update({ status: 'refunded', leituras_remaining: 0 })
      .eq('id', credit.id).eq('user_id', user.id)
  } else {
    // Partial: zera o saldo (terapeuta perde direito ao restante)
    await service.from('customer_credits')
      .update({ leituras_remaining: 0, leituras_reserved: 0 })
      .eq('id', credit.id).eq('user_id', user.id)
  }
  await service.from('credit_transactions').insert({
    user_id: user.id, credit_id: credit.id, type: 'refund',
    amount: -policy.leituras_to_refund,
    asaas_payment_id: credit.asaas_payment_id,
    notes: `D-13 arrependimento ${policy.kind}; valor R$ ${policy.value_brl.toFixed(2)}`,
  })

  await logAuditEvent({
    event_type: 'credit.refunded',
    actor_user_id: user.id, actor_email: user.email,
    target_type: 'credit', target_id: credit.id,

-- Fase 8 — Plano 08-06 — D-08 propagação de CPF do signup pro profiles via trigger.
--
-- Pré-requisitos (já aplicados pela migration 0035 do plano 08-01):
--   - profiles.cpf TEXT
--   - profiles.asaas_customer_id TEXT
--   - profiles_cpf_unique_idx (UNIQUE WHERE cpf IS NOT NULL)
--   - profiles_asaas_customer_id_unique_idx (UNIQUE WHERE asaas_customer_id IS NOT NULL)
--
-- Esta migration apenas substitui a função `public.handle_new_user()` pra ler
-- `raw_user_meta_data->>'cpf'` e gravar em `profiles.cpf` no INSERT inicial.
-- Forward-only, idempotente (create or replace).
begin;

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer set search_path = ''
as $$
begin
  insert into public.profiles (
    id, full_name, phone, cpf, specialties, tos_accepted_at, tos_version,
    subscription_status, trial_ends_at
  )
  values (
    new.id,
    coalesce(new.raw_user_meta_data->>'full_name', 'Terapeuta'),
    new.raw_user_meta_data->>'phone',
    new.raw_user_meta_data->>'cpf',   -- NOVO Fase 8 (08-06): propaga CPF assinado no signup
    case
      when jsonb_typeof(new.raw_user_meta_data->'specialties') = 'array'
      then array(
        select jsonb_array_elements_text(new.raw_user_meta_data->'specialties')
      )
      else null
    end,
    (new.raw_user_meta_data->>'tos_accepted_at')::timestamptz,
    new.raw_user_meta_data->>'tos_version',
    'trial',
    now() + interval '14 days'
  )
  on conflict (id) do nothing;

  return new;
end;
$$;

-- Re-bind trigger (idempotente; preserva nome on_auth_user_created legado da migration 0022).
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

commit;
```

import { isValidCpf } from '@/lib/auth/cpf'   // NOVO

export type TherapistGap = 'phone' | 'specialties' | 'tos' | 'cpf'   // NOVO 'cpf'

export interface TherapistProfileInput {
  phone: string | null
  specialties: string[] | null
  tos_accepted_at: string | null
  cpf: string | null   // NOVO
}

export function evaluateTherapistProfile(
  p: TherapistProfileInput,
): TherapistGateResult {
  const missing: TherapistGap[] = []
  if (!nonEmpty(p.phone)) missing.push('phone')
  if (!Array.isArray(p.specialties) || p.specialties.length < 1) missing.push('specialties')
  if (!nonEmpty(p.tos_accepted_at)) missing.push('tos')
  if (!p.cpf || !isValidCpf(p.cpf)) missing.push('cpf')   // NOVO

  if (missing.length > 0) return { status: 'incomplete', missing }
  return { status: 'ok' }
}
