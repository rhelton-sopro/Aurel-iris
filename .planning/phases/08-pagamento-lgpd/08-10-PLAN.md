```tsx
'use client'

import { useTransition } from 'react'
import { toast } from 'sonner'
import { createChargeAction } from '@/app/actions/billing'

interface Props {
  sku: 'avulsa' | 'pequeno' | 'medio' | 'grande'
  name: string
  leiturasCount: number
  priceBrl: number
  pricePerUnit: number
  savingsBrl: number  // 0 pra avulsa
  badge?: 'mais_escolhido' | 'melhor_valor' | null
}

const BADGE_LABELS: Record<NonNullable<Props['badge']>, string> = {
  mais_escolhido: 'Mais escolhido',
  melhor_valor: 'Melhor valor',
}

export function PackageCard(props: Props) {
  const [isPending, startTransition] = useTransition()

  function handleBuy() {
    startTransition(async () => {
      const r = await createChargeAction({ sku: props.sku })
      if (!r.ok) {
        toast.error(r.error)
        return
      }
      toast.success('Cobrança criada — redirecionando…')
      // Redirect pro Asaas hosted checkout
      window.location.href = r.invoice_url
    })
  }

  return (
    <div className="relative rounded-lg border-2 border-border bg-card p-5 hover:border-teal-dark/40 transition-colors">
      {props.badge ? (
        <span className="absolute -top-3 left-1/2 -translate-x-1/2 rounded-full bg-teal-dark px-3 py-0.5 text-xs font-semibold text-white">
          {BADGE_LABELS[props.badge]}
        </span>
      ) : null}

      <h3 className="text-lg font-semibold">{props.name}</h3>
      <p className="mt-1 text-sm text-muted-foreground">
        {props.leiturasCount} {props.leiturasCount === 1 ? 'leitura' : 'leituras'}
      </p>

      <div className="my-4">
        <p className="text-3xl font-bold tracking-tight">
          R$ {formatBrl(props.priceBrl)}
        </p>
        {props.leiturasCount > 1 ? (
          <p className="text-xs text-muted-foreground mt-1">
            R$ {formatBrl(props.pricePerUnit)} por leitura
          </p>
        ) : null}
        {props.savingsBrl > 0 ? (
          <p className="mt-2 text-sm font-medium text-teal-dark">
            Economia de R$ {formatBrl(props.savingsBrl)} vs Avulsa
          </p>
        ) : null}
      </div>

      <p className="text-xs text-muted-foreground mb-4">
        Validade: 12 meses após confirmação do pagamento
      </p>

      <button
        type="button"
        onClick={handleBuy}
        disabled={isPending}
        className="w-full rounded-md bg-teal-dark px-4 py-2.5 text-sm font-semibold text-white disabled:opacity-50 hover:bg-teal-dark/90"
        data-testid={`buy-${props.sku}`}
      >
        {isPending ? 'Criando cobrança…' : 'Comprar'}
      </button>
    </div>
  )
}

function formatBrl(n: number): string {
  return n.toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
}
```

**PackageGrid.tsx (server component):**

```tsx
import { PackageCard } from './PackageCard'
import { TrialCard } from './TrialCard'  // criado abaixo OU inline aqui
import type { TrialState } from '@/lib/billing/trial'

interface CreditPackage {
  id: string
  sku: 'avulsa' | 'pequeno' | 'medio' | 'grande'
  name: string
  leituras_count: number
  price_brl: number
  badge: 'mais_escolhido' | 'melhor_valor' | null
}

const AVULSA_PRICE = 99.70  // SSOT — D-02

interface Props {
  packages: CreditPackage[]
  trialState: TrialState
}

export function PackageGrid({ packages, trialState }: Props) {
  const avulsa = packages.find(p => p.sku === 'avulsa')!
  const group2 = packages.filter(p => p.sku !== 'avulsa').sort((a, b) => a.leituras_count - b.leituras_count)

  return (
    <div className="space-y-10">
      {/* Group 1: Sem compromisso */}
      <section>
        <h2 className="mb-4 text-lg font-semibold">Sem compromisso</h2>
        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
          <TrialCard state={trialState} />
          <PackageCard
            sku={avulsa.sku}
            name={avulsa.name}
            leiturasCount={avulsa.leituras_count}
            priceBrl={avulsa.price_brl}
            pricePerUnit={avulsa.price_brl}
            savingsBrl={0}
            badge={null}
          />
        </div>
      </section>

      {/* Group 2: Pacotes com economia */}
      <section>
        <h2 className="mb-4 text-lg font-semibold">Pacotes com economia</h2>
        <div className="grid grid-cols-1 gap-4 md:grid-cols-3">
          {group2.map(pkg => {
            const pricePerUnit = pkg.price_brl / pkg.leituras_count
            const savingsPerUnit = AVULSA_PRICE - pricePerUnit
            const totalSavings = savingsPerUnit * pkg.leituras_count
            return (
              <PackageCard
                key={pkg.id}
                sku={pkg.sku}
                name={pkg.name}
                leiturasCount={pkg.leituras_count}
                priceBrl={pkg.price_brl}
                pricePerUnit={pricePerUnit}
                savingsBrl={totalSavings}
                badge={pkg.badge}
              />
            )
          })}
        </div>
      </section>
    </div>
  )
}
```

**page.tsx (server component):**

```tsx
import { redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { PackageGrid } from '@/components/billing/PackageGrid'
import { getTrialState } from '@/lib/billing/trial'
import { DisclaimerCopy } from '@/components/legal/DisclaimerCopy'

export const metadata = { title: 'Comprar créditos — Iris Codex' }

export default async function ComprarPage() {
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) redirect('/login?next=/assinatura/comprar')

  const [{ data: packages }, trialState] = await Promise.all([
    supabase.from('credit_packages')
      .select('id, sku, name, leituras_count, price_brl, badge, display_order')
      .eq('active', true)
      .order('display_order'),
    getTrialState(user.id),
  ])

  return (
    <div className="max-w-5xl mx-auto px-4 py-8 space-y-8">
      <header className="text-center space-y-2">
        <h1 className="text-2xl font-bold">Comprar créditos</h1>
        <p className="text-sm text-muted-foreground">
          Pague apenas pelo que vai usar. Sem assinatura mensal. Créditos válidos por 12 meses.
        </p>
      </header>

      {packages && packages.length > 0 ? (
        <PackageGrid packages={packages as never} trialState={trialState} />
      ) : (
        <p className="text-center text-muted-foreground">
          Pacotes em atualização. Tente novamente em alguns instantes.
        </p>
      )}

      <section className="mt-10 rounded-md bg-muted/30 p-5 text-sm space-y-2">
        <h3 className="font-semibold">Informações importantes</h3>
        <ul className="list-disc pl-6 space-y-1 text-muted-foreground">
          <li>Pagamento via PIX, cartão de crédito ou boleto (checkout Asaas).</li>
          <li>Validade: 12 meses a partir da confirmação do pagamento.</li>
          <li>Direito de arrependimento: 7 dias (reembolso integral se nenhuma leitura foi usada).
            Detalhes em <a href="/termos#arrependimento" className="underline">Termos de Uso</a>.</li>
          <li>Nota fiscal emitida automaticamente após confirmação.</li>
        </ul>
      </section>

      <DisclaimerCopy variant="footer" />
    </div>
  )
}
```
