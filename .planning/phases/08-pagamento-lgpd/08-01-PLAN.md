2. `fifo_reserve_credit(p_user_id uuid, p_reading_id uuid) returns table(reservation_id uuid, credit_id uuid, source text)` — RESEARCH §FIFO Consumption SQL lines 466-503 adaptado pra PLPGSQL com return + advisory-lock-on-user-id pra fechar race condition do pitfall #3 (SELECT FOR UPDATE em row inexistente não bloqueia):

```sql
create or replace function public.fifo_reserve_credit(
  p_user_id uuid,
  p_reading_id uuid
) returns table(reservation_id uuid, credit_id uuid, source text)
language plpgsql security definer as $$
declare
  v_internal boolean;
  v_in_trial boolean;
  v_credit_id uuid;
  v_reservation_id uuid;
begin
  -- Advisory lock no user_id pra serializar reserves do mesmo terapeuta
  perform pg_advisory_xact_lock(hashtext(p_user_id::text));

  -- 1. Internal_use bypass (D-09): cria reservation com credit_id NULL, source='internal'
  select internal_use into v_internal from public.profiles where id = p_user_id;
  if v_internal then
    insert into public.credit_reservations (user_id, credit_id, reading_id, expires_at)
    values (p_user_id, null, p_reading_id, now() + interval '7 days')
    returning id into v_reservation_id;
    return query select v_reservation_id, null::uuid, 'internal'::text;
    return;
  end if;

  -- 2. Trial check (D-06)
  v_in_trial := public.is_in_trial(p_user_id);
  if v_in_trial then
    -- Atomic increment: update + return; CHECK constraint barra se >= max
    update public.trial_status
       set trial_readings_used = trial_readings_used + 1
     where user_id = p_user_id
       and trial_readings_used < trial_readings_max;
    if not found then
      raise exception 'no_balance' using errcode = 'P0001';
    end if;
    insert into public.credit_reservations (user_id, credit_id, reading_id, expires_at)
    values (p_user_id, null, p_reading_id, now() + interval '7 days')
    returning id into v_reservation_id;
    insert into public.credit_transactions (user_id, credit_id, reading_id, type, amount)
    values (p_user_id, null, p_reading_id, 'reserve', -1);
    return query select v_reservation_id, null::uuid, 'trial'::text;
    return;
  end if;

  -- 3. FIFO consume — lock oldest active credit with available balance
  select id into v_credit_id
    from public.customer_credits
   where user_id = p_user_id
     and status = 'active'
     and expires_at > now()
     and (leituras_remaining - leituras_reserved) > 0
   order by purchase_date asc
   limit 1
   for update;

  if v_credit_id is null then
    raise exception 'no_balance' using errcode = 'P0001';
  end if;

  update public.customer_credits
     set leituras_reserved = leituras_reserved + 1
   where id = v_credit_id;

  insert into public.credit_reservations (user_id, credit_id, reading_id, expires_at)
  values (p_user_id, v_credit_id, p_reading_id, now() + interval '7 days')
  returning id into v_reservation_id;

  insert into public.credit_transactions (user_id, credit_id, reading_id, type, amount)
  values (p_user_id, v_credit_id, p_reading_id, 'reserve', -1);

  return query select v_reservation_id, v_credit_id, 'credit'::text;
end;
$$;

grant execute on function public.fifo_reserve_credit(uuid, uuid) to authenticated;
```

3. `release_reservation(p_reading_id uuid, p_reason text) returns boolean` — pra cron daily + cancel manual (idempotent via status guard):

```sql
create or replace function public.release_reservation(
  p_reading_id uuid,
  p_reason text default 'manual'
) returns boolean
language plpgsql security definer as $$
declare
  v_reservation public.credit_reservations%rowtype;
begin
  update public.credit_reservations
     set status = case when p_reason = 'expired' then 'expired' else 'released' end,
         released_at = now()
   where reading_id = p_reading_id and status = 'active'
   returning * into v_reservation;
  if not found then return false; end if;

  -- Refund saldo: credit OR trial
  if v_reservation.credit_id is not null then
    update public.customer_credits
       set leituras_reserved = greatest(leituras_reserved - 1, 0)
     where id = v_reservation.credit_id;
  else
    -- trial — devolve readings_used
    update public.trial_status
       set trial_readings_used = greatest(trial_readings_used - 1, 0)
     where user_id = v_reservation.user_id;
  end if;

  insert into public.credit_transactions (user_id, credit_id, reading_id, type, amount, notes)
  values (v_reservation.user_id, v_reservation.credit_id, p_reading_id,
          case when p_reason = 'expired' then 'expire' else 'release' end,
          1, p_reason);
  return true;
end;
$$;

grant execute on function public.release_reservation(uuid, text) to authenticated;
```
