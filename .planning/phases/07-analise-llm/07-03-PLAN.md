```typescript
/**
 * Anthropic SDK client — server-only factory.
 *
 * - Module-scope singleton (fresh request reuses connection pool).
 * - MODEL via env var (D-T2) — allows opus override in test/staging.
 * - DEFAULT_SYSTEM_BLOCK_ARGS includes `cache_control: { type: 'ephemeral' }`
 *   to enable Anthropic prompt caching (5-min TTL, ~90% read discount).
 *   Pitfall 4 (07-RESEARCH.md): silently no-cache if system prompt < 2048
 *   tokens — `lib/anthropic/prompts.ts` token-counts at module init and
 *   logs WARN if below threshold.
 *
 * Phase 7 | Plan 07-03 | Decisions: D-T2, RESEARCH Pitfall 4
 */
import 'server-only'
import Anthropic from '@anthropic-ai/sdk'

const apiKey = process.env.ANTHROPIC_API_KEY
if (!apiKey) {
  throw new Error(
    '[lib/anthropic/client] ANTHROPIC_API_KEY env var is required. ' +
      'Set in apps/web/.env.local for dev or Vercel Environment Variables for prod.',
  )
}

/** Module-scope client — reuses HTTP connection pool. */
export const anthropicClient = new Anthropic({ apiKey })

/**
 * Model snapshot ID. Default `claude-sonnet-4-6` (dateless alias = literal ID).
 * Override via `ANTHROPIC_MODEL` env var for test/staging (D-T2).
 */
export const MODEL = process.env.ANTHROPIC_MODEL ?? 'claude-sonnet-4-6'

/**
 * Default args for the system block in `messages.stream({...})`.
 * `cache_control: { type: 'ephemeral' }` enables Anthropic-side prompt caching.
 *
 * Sonnet 4.6 cache threshold: 2048 input tokens. `lib/anthropic/prompts.ts`
 * loads system.md at init and warns if below threshold.
 */
export const DEFAULT_SYSTEM_CACHE_CONTROL = { type: 'ephemeral' as const }

/** Hard ceiling on output tokens per analysis (D-S3 timeout aligned). */
export const MAX_OUTPUT_TOKENS = 16000

/** Anthropic Sonnet 4.6 pricing (USD per 1M tokens) for cost telemetry. */
export const PRICING_SONNET_4_6 = {
  input_per_mtok: 3.0,
  output_per_mtok: 15.0,
  cache_write_per_mtok: 3.75,
  cache_read_per_mtok: 0.3,
} as const

/** Compute cost in USD given Anthropic usage object (Pitfall 4 — track all 4 buckets). */
export function estimateCostUsd(usage: {
  input_tokens?: number
  output_tokens?: number
  cache_creation_input_tokens?: number
  cache_read_input_tokens?: number
}): number {
  const i = (usage.input_tokens ?? 0) / 1_000_000
  const o = (usage.output_tokens ?? 0) / 1_000_000
  const cw = (usage.cache_creation_input_tokens ?? 0) / 1_000_000
  const cr = (usage.cache_read_input_tokens ?? 0) / 1_000_000
  return (
    i * PRICING_SONNET_4_6.input_per_mtok +
    o * PRICING_SONNET_4_6.output_per_mtok +
    cw * PRICING_SONNET_4_6.cache_write_per_mtok +
    cr * PRICING_SONNET_4_6.cache_read_per_mtok
  )
}
```

**Sub-task 3a — `apps/web/lib/anthropic/prompts.ts`:**

```typescript
/**
 * Prompt loader + mustache substitution for Phase 7 LLM analysis.
 *
 * Files: `apps/web/prompts/system.md` + `apps/web/prompts/feature-injection.md`.
 * Both are LITERAL copies of SPEC.md §6 (D-PR1 frozen contract — no rewrite).
 *
 * Vercel deploy: `next.config.ts` `outputFileTracingIncludes` traces
 * `prompts/**\/*` into the analyze route bundle (Pitfall 9 RESEARCH). Without
 * that config, prod 500s with ENOENT.
 *
 * Token count: `loadSystemPrompt()` warns at module init if system.md < 2200
 * tokens (Sonnet 4.6 cache_control threshold 2048 + margem). Below threshold
 * means cache_control is silently disabled, costs jump ~10×.
 *
 * Phase 7 | Plan 07-03 | Decisions: D-PR1, RESEARCH Pitfall 4 + Pitfall 9
 */
import 'server-only'
import { readFileSync } from 'node:fs'
import path from 'node:path'

let _systemCache: string | null = null
let _injectionCache: string | null = null

const PROMPTS_DIR = path.join(process.cwd(), 'prompts')

export function loadSystemPrompt(): string {
  if (_systemCache !== null) return _systemCache
  const filepath = path.join(PROMPTS_DIR, 'system.md')
  _systemCache = readFileSync(filepath, 'utf8')
  // Pitfall 4 — token-count check at first load.
  // Approximation: 1 token ≈ 4 chars for pt-BR + markdown structure.
  // For Anthropic accuracy, run client.messages.countTokens at deploy-smoke.
  const estimatedTokens = Math.ceil(_systemCache.length / 4)
  if (estimatedTokens < 2200) {
    console.warn(
      `[lib/anthropic/prompts] system.md is ~${estimatedTokens} tokens, ` +
        `below 2200 margin (Sonnet 4.6 cache_control threshold = 2048). ` +
        `Prompt caching may be silently disabled. Pitfall 4 RESEARCH.`,
    )
  }
  return _systemCache
}

export function loadInjectionTemplate(): string {
  if (_injectionCache !== null) return _injectionCache
  const filepath = path.join(PROMPTS_DIR, 'feature-injection.md')
  _injectionCache = readFileSync(filepath, 'utf8')
  return _injectionCache
}

/**
 * Mustache-style substitution: `{{name}}` → vars[name] || ''.
 * Missing keys substitute to empty string (not literal `{{name}}`) so
 * unrendered placeholders don't leak to the LLM as instructions.
 */
export function renderInjection(template: string, vars: Record<string, string>): string {
  return template.replace(/\{\{([\w_]+)\}\}/g, (_, key) => vars[key] ?? '')
}

/** Reset module cache — TEST ONLY. Production callers must NOT call this. */
export function _resetPromptsCache(): void {
  _systemCache = null
  _injectionCache = null
}
```
