"""PDF + DOCX extractors for the RAG ingestion pipeline (D-C4).

Lazy imports keep test_persist.py importable without installing PyMuPDF
(matches `import httpx  # lazy` pattern in vision-service/modal_app.py:94).

Phase: 06-rag-ingestao | Plan: 06-04 | Decisions: D-C4, RESEARCH lines 290–335
"""
from __future__ import annotations

import functools
from pathlib import Path
from typing import Any


def is_scanned_page(page: Any) -> bool:
    """Return True when page has no extractable text and is dominated by an image.

    RESEARCH lines 314–333 verbatim. Used to flag books needing OCR (D-S deferred).
    """
    text = page.get_text("text").strip()
    if text:
        # Some pages contain replacement chars from invalid Unicode → treat as scan
        if text.count("�") / max(len(text), 1) > 0.3:
            return True
        return False
    images = page.get_images()
    if not images:
        return True  # blank page
    page_area = page.rect.width * page.rect.height
    for img in images:
        for r in page.get_image_rects(img[0]):
            if (r.width * r.height) / page_area > 0.95:
                return True
    return False


def extract_pdf(path: Path, extractor: str = "pymupdf") -> list[dict]:
    """Return list of {page, text, scan_detected} per page.

    Args:
        path:      .pdf file location.
        extractor: "pymupdf" (D-C4 primary) or "pdfplumber" (D-C4 fallback).

    Returns:
        list[dict] — one entry per page, ordered by page number (1-indexed).
    """
    if extractor == "pymupdf":
        return _extract_pdf_pymupdf(path)
    if extractor == "pdfplumber":
        return _extract_pdf_pdfplumber(path)
    raise ValueError(f"Unknown extractor: {extractor!r}")


def _extract_pdf_pymupdf(path: Path) -> list[dict]:
    import pymupdf  # lazy
    doc = pymupdf.open(str(path))
    pages: list[dict] = []
    try:
        for page_num, page in enumerate(doc):
            text = page.get_text("text")
            scan = is_scanned_page(page)
            pages.append({
                "page": page_num + 1,
                "text": text,
                "scan_detected": scan,
            })
    finally:
        doc.close()
    return pages


def _extract_pdf_pdfplumber(path: Path) -> list[dict]:
    import pdfplumber  # lazy
    pages: list[dict] = []
    with pdfplumber.open(str(path)) as doc:
        for page_num, page in enumerate(doc.pages):
            text = page.extract_text() or ""
            # pdfplumber doesn't expose scan-detection cheaply; treat empty text
            # with at least one image as scan
            images = page.images or []
            scan = (not text.strip()) and bool(images)
            pages.append({
                "page": page_num + 1,
                "text": text,
                "scan_detected": scan,
            })
    return pages


def extract_docx(path: Path) -> list[dict]:
    """One synthetic 'page' with the whole text — DOCX has no real page boundary."""
    import docx2txt  # lazy
    text = docx2txt.process(str(path))
    return [{"page": None, "text": text, "scan_detected": False}]
```

Target 500 tokens, flex 300–700, overlap 80, NO overlap across chapter/section.

content_hash canonicalization is LOCKED to sha256(text.strip().encode('utf-8')) —
no NFC/NFD normalization, no lowercase, no whitespace collapse. Test asserts
hash of "hello" against the well-known sha256 hex digest.

Phase: 06-rag-ingestao | Plan: 06-04 | Decisions: D-C1, D-C2, D-C3, D-E2,

"""
from __future__ import annotations

import functools
import hashlib
import re
from dataclasses import dataclass

# D-C1 locked params
TARGET_TOKENS = 500
MIN_TOKENS = 300
MAX_TOKENS = 700
OVERLAP_TOKENS = 80

# D-C2 hierarchy detection — multilingual chapter/section markers
CHAPTER_RE = re.compile(
    r"^\s*(?:CAP[IÍ]TULO|CHAPTER|CAPITOLO)\s+[IVXLCDM\d]+",
    re.IGNORECASE | re.MULTILINE,
)
SECTION_RE = re.compile(
    r"^\s*(?:\d+\.\d+(?:\.\d+)?\s+\S|[A-Z][A-Z\s]{4,})\s*$",
    re.MULTILINE,
)


@dataclass
class Chunk:
    text: str
    chapter: str | None
    section: str | None
    page: int | None
    tokens_estimated: int


@functools.lru_cache(maxsize=1)
def _get_encoder():
    import tiktoken  # lazy
    return tiktoken.get_encoding("cl100k_base")


def count_tokens(text: str) -> int:
    """tiktoken cl100k_base length — proxy for Voyage tokens (under-counts ~1.2×)."""
    return len(_get_encoder().encode(text))

**Chunker algorithm** (executor implements this — RESEARCH/PATTERNS gave the skeleton; this is the concrete logic):

1. **Pass 1 — assemble linear text-with-marker stream:** iterate pages; for each page's text, scan for CHAPTER_RE matches → emit `("CHAPTER_BOUNDARY", chapter_text)`; scan for SECTION_RE → emit `("SECTION_BOUNDARY", section_text)`; else emit `("TEXT", paragraph, page_num)`. Split text by `\n\n+` to get paragraph units.
2. **Pass 2 — group paragraphs into chunks per (chapter, section) tuple:**
   - Maintain `current_chapter`, `current_section`, `buffer = []`, `buffer_tokens = 0`.
   - On `CHAPTER_BOUNDARY`: flush `buffer` as a final Chunk for the previous (chapter, section) — even if buffer < MIN_TOKENS, emit it (do not span chapters). Update `current_chapter`. Reset `current_section = None`.
   - On `SECTION_BOUNDARY`: flush buffer (no overlap when crossing sections, per D-C1 last clause). Update `current_section`.
   - On `TEXT (paragraph, page)`: tokens = count_tokens(paragraph). If `buffer_tokens + tokens > MAX_TOKENS`: flush buffer as Chunk(...); rewind by `OVERLAP_TOKENS` worth of TRAILING text from the just-flushed buffer to seed next chunk (within-section overlap only). Append paragraph.
3. **Final flush:** emit any non-empty buffer as Chunk.
4. **Post-pass — enforce MIN_TOKENS:** any chunk with `tokens_estimated < MIN_TOKENS` is preserved if it's the LAST chunk of a section/chapter (boundary case acceptable per D-C1 flex); else merge with predecessor in same (chapter, section) group.

**Edge cases tested in `test_chunker.py`:**
- All chunks within `[300, 700]` tokens (with the section-tail boundary exception above)
- No two adjacent chunks share text when their `chapter` differs
- Within a single section, adjacent chunks share approximately 80 tokens (test asserts ≥60 token overlap, ≤120 tokens — flex tolerance)
- Each chunk has `chapter`, `section`, `page`, `tokens_estimated` populated (page = first page seen during accumulation)

def chunk_book(pages: list[dict], book_meta: dict) -> list[Chunk]:
    """See module docstring."""
    # Pass 1 — build linear stream
    stream: list[tuple[str, ...]] = []
    for page_dict in pages:
        page_num = page_dict.get("page")
        text = page_dict.get("text", "") or ""
        # Detect chapter/section markers within page text via regex
        # Split text into paragraphs by \n\n+
        for para in re.split(r"\n\n+", text):
            stripped = para.strip()
            if not stripped:
                continue
            chapter_m = CHAPTER_RE.search(stripped)
            section_m = SECTION_RE.search(stripped)
            if chapter_m:
                stream.append(("CHAPTER", chapter_m.group(0).strip()))
            elif section_m:
                stream.append(("SECTION", section_m.group(0).strip()))
            else:
                stream.append(("TEXT", stripped, page_num))

    # Pass 2 — chunk-build
    chunks: list[Chunk] = []
    current_chapter: str | None = None
    current_section: str | None = None
    buffer_paragraphs: list[tuple[str, int | None]] = []  # [(text, page)]
    buffer_tokens = 0
    first_page: int | None = None

    def flush(*, allow_below_min: bool = False) -> None:
        nonlocal buffer_paragraphs, buffer_tokens, first_page
        if not buffer_paragraphs:
            return
        chunk_text = "\n\n".join(p[0] for p in buffer_paragraphs)
        chunks.append(Chunk(
            text=chunk_text,
            chapter=current_chapter,
            section=current_section,
            page=first_page,
            tokens_estimated=count_tokens(chunk_text),
        ))
        buffer_paragraphs = []
        buffer_tokens = 0
        first_page = None

    for entry in stream:
        kind = entry[0]
        if kind == "CHAPTER":
            flush(allow_below_min=True)  # boundary: emit even if small
            current_chapter = entry[1]
            current_section = None
        elif kind == "SECTION":
            flush(allow_below_min=True)
            current_section = entry[1]
        else:  # TEXT
            _, paragraph, page = entry
            tokens = count_tokens(paragraph)
            if buffer_tokens + tokens > MAX_TOKENS and buffer_tokens >= MIN_TOKENS:
                # Within-section overlap: keep last ~OVERLAP_TOKENS of buffer
                tail_tokens = 0
                tail: list[tuple[str, int | None]] = []
                for p, pg in reversed(buffer_paragraphs):
                    pt = count_tokens(p)
                    if tail_tokens + pt > OVERLAP_TOKENS:
                        break
                    tail.insert(0, (p, pg))
                    tail_tokens += pt
                flush()
                buffer_paragraphs = list(tail)
                buffer_tokens = tail_tokens
                if tail:
                    first_page = tail[0][1]
            if first_page is None:
                first_page = page
            buffer_paragraphs.append((paragraph, page))
            buffer_tokens += tokens
    flush(allow_below_min=True)
    return chunks
```
