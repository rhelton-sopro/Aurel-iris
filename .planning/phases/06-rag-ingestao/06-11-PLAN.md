/**
 * D-N2 reranker — voyage-rerank-2.5 with graceful fallback.
 *
 * Returns `args.candidates.slice(0, topK)` (cosine sort) on any error. NEVER
 * throws. RESEARCH §Reranking: free tier 200M tokens covers ~12+ months of
 * dogfooding; latency ~600ms p50.
 *
 * Phase: 06-rag-ingestao | Plan: 06-11 | Decisions: D-N2, D-N4
 */
import 'server-only'
import { VoyageAIClient } from 'voyageai'
import type { KnowledgeChunkRow } from './types'

export const RERANK_MODEL = process.env.VOYAGE_RERANK_MODEL ?? 'voyage-rerank-2.5'

export interface RerankArgs {
  query: string
  candidates: KnowledgeChunkRow[]
  /** Default 30 per D-R3 cap. */
  topK?: number
}

/**
 * Reorder candidates by reranker relevance. Graceful fallback semantics —
 * D-N2: "se rerank API falha, retorna top-30 cosine puro (não derruba retrieval)".
 */
export async function rerankChunks(args: RerankArgs): Promise<KnowledgeChunkRow[]> {
  const topK = args.topK ?? 30
  const apiKey = process.env.VOYAGE_API_KEY
  if (!apiKey) {
    console.warn('[rag.rerank] VOYAGE_API_KEY missing — falling back to cosine sort')
    return args.candidates.slice(0, topK)
  }
  if (args.candidates.length === 0) {
    return []
  }
  try {
    const client = new VoyageAIClient({ apiKey })
    const result = await client.rerank({
      query: args.query,
      documents: args.candidates.map((c) => c.text),
      model: RERANK_MODEL,
      topK,
    })
    // Reorder by reranker indices, replace score with rerank relevanceScore
    return result.data.map((r) => ({
      ...args.candidates[r.index],
      score: r.relevanceScore,
    }))
  } catch (err) {
    console.warn('[rag.rerank] reranker failed — falling back to cosine sort:', err)
    return args.candidates.slice(0, topK)
  }
}

/**
 * `retrieveRelevantKnowledge` — server action exposing the RAG retrieval pipeline.
 *
 * Composition (D-R1..R5, D-N2, D-N4):
 *   1. Auth gate (Supabase server-side user check)
 *   2. Build queries (Family A + Family B per D-R2)
 *   3. Embed all queries in parallel (Promise.all — D-R5)
 *   4. Run pgvector RPC `match_knowledge_chunks` for each embedding (Promise.all)
 *   5. Dedup by id, keep best cosine score
 *   6. Rerank top-50 → top-30 via voyage-rerank-2.5 (graceful fallback — D-N2)
 *   7. Apply D-R4 weights (clinical_data 1.5×, alta_prioridade 1.1×, dimensoes intersect 1.2×)
 *   8. Final sort by score desc, slice(0, 30)
 *   9. Telemetry log (no PII)
 *
 * Phase: 06-rag-ingestao | Plan: 06-11 | Decisions: D-R1, D-R3, D-R4, D-R5, D-N2
 */
'use server'
import { createClient } from '@/lib/supabase/server'
import { embedTexts } from './embed'
import { buildFamilyA, buildFamilyB, type IrisFeaturesForRag } from './build-queries'
import { rerankChunks } from './rerank'
import { applyWeights } from './score-weights'
import type { KnowledgeChunkRow, ReportSection } from './types'

const TOP_K_PER_QUERY = 10        // D-N2: bumped from 5 → 10 to feed reranker (overfetch)
const FINAL_CAP = 30              // D-R3 cap
const MATCH_THRESHOLD = 0.0       // RPC default — no minimum score floor

export interface RetrieveArgs {
  features: IrisFeaturesForRag
  reportSections: ReportSection[]
}

/**
 * Hardcoded set of book names marked alta_prioridade in books_manifest.json (06-03 founder gate).
 * If founder updates the manifest, this set must be updated in lockstep — RESEARCH Pitfall 9
 * adapted to manifest sync. A future plan may load this dynamically via a server-side cache;
 * for v1, hardcoded keeps the retrieval cold-start fast and avoids a DB call per request.
 *
 * Source of truth: vision-service/scripts/data/books_manifest.json — entries where
 * `alta_prioridade: true`. Update this constant when founder bumps that flag.
 */
const ALTA_PRIORIDADE_BOOKS: ReadonlySet<string> = new Set([
  // 06-03 founder defaults — update if founder gate output differed
  'Bernard Jensen Iridology Simplified',
  'dictionary of iridology pdf',
  'Iridologia Del Profondo Birello Lucio Rito Daniele Lo 2007 Enea Edizioni 84f083031f5e812f466e932 1',
  'What the Eye Reveals',
  'A Iridologia Em Defesa Da Vida',
  // Note: book key naming follows manifest_assist.canonical_book_key() — strip leading digits, replace dashes with spaces.
])

export async function retrieveRelevantKnowledge(
  args: RetrieveArgs,
): Promise<KnowledgeChunkRow[]> {
  const supabase = await createClient()
  const { data: { user }, error: authErr } = await supabase.auth.getUser()
  if (!user || authErr) {
    throw new Error('Unauthenticated')
  }

  // Step 2: build queries
  const queriesA = buildFamilyA(args.features)
  const queriesB = buildFamilyB(args.features, args.reportSections)
  const allQueries = [...queriesA, ...queriesB]
  if (allQueries.length === 0) {
    return []
  }

  // Step 3: embed queries in parallel (D-R5)
  const { embeddings } = await embedTexts({ texts: allQueries, inputType: 'query' })

  // Step 4: pgvector RPC for each query, parallel (D-R5)
  const rpcPromises = embeddings.map((emb) =>
    supabase.rpc('match_knowledge_chunks', {
      query_embedding: emb as unknown as string,  // SDK serializes number[] to vector format
      match_count: TOP_K_PER_QUERY,
      match_threshold: MATCH_THRESHOLD,
    }),
  )
  const responses = await Promise.all(rpcPromises)

  // Step 5: dedup by id, keep best cosine score
  const dedupMap = new Map<string, KnowledgeChunkRow>()
  for (const { data } of responses) {
    if (!data) continue
    for (const row of data as Array<Record<string, unknown>>) {
      const id = String(row.id)
      const score = Number(row.score)
      const existing = dedupMap.get(id)
      if (!existing || score > existing.score) {
        // Cast: RPC return shape matches KnowledgeChunkRow modulo metadata jsonb cast
        dedupMap.set(id, {
          id,
          text: String(row.content),
          source_book: String(row.source_book),
          chapter: row.source_chapter == null ? null : String(row.source_chapter),
          section: null,  // section lives inside metadata, not its own column
          page: row.source_page == null ? null : Number(row.source_page),
          metadata: row.metadata as KnowledgeChunkRow['metadata'],
          source_type: (row.source_type as 'biblioteca' | 'clinical_data') ?? 'biblioteca',
          score,
        })
      }
    }
  }
  let candidates = Array.from(dedupMap.values())

  // Step 6: rerank (graceful fallback — D-N2)
  candidates = await rerankChunks({
    query: allQueries.join(' '),
    candidates,
    topK: FINAL_CAP * 2,  // overfetch for weight-driven re-ordering
  })

  // Step 7: apply D-R4 weights
  const primarySection = args.reportSections[0] ?? null
  candidates = applyWeights(candidates, primarySection, ALTA_PRIORIDADE_BOOKS as Set<string>)

  // Step 8: final sort + cap
  candidates.sort((a, b) => b.score - a.score)
  const result = candidates.slice(0, FINAL_CAP)

  // Step 9: telemetry (no PII — only counts and scores)
  console.info({
    event: 'rag_retrieve',
    queries_count: allQueries.length,
    chunks_returned: result.length,
    top_score: result[0]?.score ?? null,
    bottom_score: result[result.length - 1]?.score ?? null,
  })
  return result
}
