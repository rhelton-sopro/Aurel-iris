New `apps/web/scripts/audit-vocabulary-db.mjs` (full file):

```javascript
#!/usr/bin/env node
// scripts/audit-vocabulary-db.mjs
// LGPD audit for the metadata.tags_livres field of knowledge_chunks.
// RESEARCH Pitfall 6: chunk content may quote forbidden words verbatim from books
// (allowed); but tags WE write must NEVER include them.
//
// Phase: 06-rag-ingestao | Plan: 06-12 | Decisions: D-T6, RESEARCH Pitfall 6

import { createClient } from '@supabase/supabase-js'

const SUPABASE_URL = process.env.SUPABASE_URL || process.env.NEXT_PUBLIC_SUPABASE_URL
const SUPABASE_SERVICE_ROLE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY

if (!SUPABASE_URL) {
  console.error('SUPABASE_URL is not set')
  process.exit(2)
}
if (!SUPABASE_SERVICE_ROLE_KEY) {
  console.error('SUPABASE_SERVICE_ROLE_KEY is not set')
  process.exit(2)
}
if (!SUPABASE_SERVICE_ROLE_KEY.startsWith('eyJ')) {
  console.error('SUPABASE_SERVICE_ROLE_KEY appears malformed (no eyJ prefix)')
  process.exit(2)
}

const client = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, {
  auth: { persistSession: false, autoRefreshToken: false },
})

// Pattern matches the same words as the file-scan audits.
// We scan the JSONB-as-text representation; the DB ILIKE handles word boundaries
// approximately (no \b in PostgreSQL ILIKE). The Python/JS file audits use \b regex,
// so the DB scan is slightly broader (may yield false positives that need founder review).
const FORBIDDEN_TERMS = ['diagnóstico', 'tratamento', 'cura']

async function main() {
  const hits = []
  for (const term of FORBIDDEN_TERMS) {
    // Cast metadata to text and ILIKE — case-insensitive, accent-aware
    const { data, error } = await client
      .from('knowledge_chunks')
      .select('id, source_book, metadata')
      .filter('metadata::text', 'ilike', `%${term}%`)
      .limit(500)
    if (error) {
      console.error(`Query failed for term "${term}":`, error.message)
      process.exit(2)
    }
    for (const row of data || []) {
      // Only flag if the term appears in tags_livres (written by us), not in
      // the chunk content body. The query above scans the entire metadata JSON,
      // so we must narrow here.
      const tagsLivres = row.metadata?.tags_livres || []
      if (Array.isArray(tagsLivres) && tagsLivres.some((t) => String(t).toLowerCase().includes(term.toLowerCase()))) {
        hits.push({
          id: row.id,
          source_book: row.source_book,
          term,
          tags_livres: tagsLivres,
        })
      }
    }
  }

  if (hits.length === 0) {
    console.log(`OK: zero hits across ${FORBIDDEN_TERMS.join(', ')} in metadata.tags_livres`)
    process.exit(0)
  }
  console.error(`VOCAB FAIL — ${hits.length} chunk(s) have forbidden vocab in tags_livres:`)
  for (const h of hits) {
    console.error(`  id=${h.id} | source_book="${h.source_book}" | term="${h.term}" | tags_livres=${JSON.stringify(h.tags_livres)}`)
  }
  process.exit(1)
}

main().catch((err) => {
  console.error('FATAL:', err)
  process.exit(2)
})
```

**Note on DB query approach:** the script scans `metadata::text ILIKE '%X%'` first (broader) and then narrows to actual `tags_livres` array contents. Two-step pattern reduces false negatives if the JSON structure changes — broad scan + narrow filter is safer than just filtering by jsonb path. Founder reviews any hit; if the hit is a `tags_livres` entry that's a verbatim quote of a forbidden word INSIDE a longer string (e.g., `"diagnóstico do Birello"`), founder rewrites the tag.

**Important env detail for `audit:vocabulary:db`:** The script needs `SUPABASE_SERVICE_ROLE_KEY`. Founder runs locally; CI does NOT run this audit (CI doesn't have service-role access by design). For CI, the file-scan audits are sufficient.
