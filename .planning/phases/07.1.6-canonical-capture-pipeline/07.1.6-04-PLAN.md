    ```typescript
    /**
     * POST /api/capture/canonicalize
     *
     * Server-side canonicalize endpoint (Phase 07.1.6). Único entry point para
     * canonical capture pipeline:
     *   - finalizeReadingAction (Wave 2 Plan 05) — chama via internal fetch após 6/6 uploads
     *   - /admin/calibration "Re-canonicalizar" button (Wave 2 Plan 06) — chama via client fetch
     *
     * Idempotente (D-05): re-chamar com mesmo readingId regenera os canonical crops
     * sobre os originais inalterados; canonical_storage_path é overwrite via service-role.
     *
     * Contract:
     *   POST { "readingId": "<uuid>" }
     *   200 { "results": CanonicalizeResult[], "metadata": CanonicalMetadata, "status_summary": {...} }
     *   400 — body inválido / readingId ausente
     *   401 — sem sessão Supabase
     *   404 — reading não pertence ao user
     *   500 — falha não-recuperável no orchestrator (Sonnet API down, Storage down, etc.)
     *
     * D-01 enforcement: o endpoint NÃO retorna 500 quando alguns crops caem em
     * 'fallback' — o orchestrator (canonicalizeReading) absorve falhas per-imagem
     * e devolve um CanonicalizeResult com canonical_status='fallback'. Endpoint
     * só retorna 500 em falha sistêmica (Sonnet API totalmente down, Storage 5xx
     * em todas as imagens, etc.).
     */
    import 'server-only'
    import { NextResponse, type NextRequest } from 'next/server'
    import { createClient } from '@/lib/supabase/server'
    import { canonicalizeReading } from '@/lib/canonicalize'

    export const runtime = 'nodejs'

    interface CanonicalizeRequestBody {
      readingId?: unknown
    }

    export async function POST(request: NextRequest): Promise<NextResponse> {
      // 1. Parse body
      let body: CanonicalizeRequestBody
      try {
        body = (await request.json()) as CanonicalizeRequestBody
      } catch {
        return NextResponse.json({ error: 'Invalid JSON body' }, { status: 400 })
      }
      if (typeof body.readingId !== 'string' || body.readingId.length === 0) {
        return NextResponse.json({ error: 'readingId (string) required' }, { status: 400 })
      }
      const readingId = body.readingId

      // 2. Auth gate (user session — RLS-enforced client)
      const supabase = await createClient()
      const { data: { user }, error: authError } = await supabase.auth.getUser()
      if (!user || authError) {
        return NextResponse.json({ error: 'Unauthenticated' }, { status: 401 })
      }

      // 3. Ownership check (RLS via user client; .eq('therapist_id', user.id) is defense-in-depth)
      const { data: reading, error: readingError } = await supabase
        .from('readings')
        .select('id, therapist_id')
        .eq('id', readingId)
        .eq('therapist_id', user.id)
        .single()
      if (readingError || !reading) {
        return NextResponse.json({ error: 'Reading not found' }, { status: 404 })
      }

      // 4. Delegate to orchestrator (service-role; bypasses RLS for canonical writes)
      try {
        const { results, metadata } = await canonicalizeReading(readingId, user.id)
        return NextResponse.json({
          results,
          metadata,
          status_summary: metadata.status_summary,
        }, { status: 200 })
      } catch (err) {
        const errorMessage = err instanceof Error ? `${err.name}: ${err.message}` : String(err)
        console.error(
          `[api/capture/canonicalize] error reading=${readingId}:`,
          errorMessage,
          err instanceof Error && err.stack ? `\n${err.stack}` : '',
        )
        return NextResponse.json({ error: errorMessage }, { status: 500 })
      }
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser/internal-fetch → POST /api/capture/canonicalize | Untrusted input: `readingId` field. Tampering risk: user tries another user's readingId. |
| API route → canonicalizeReading (service-role) | RLS bypass internal. Inputs already validated by ownership gate. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07.1.6-16 | Spoofing | request body `readingId` | mitigate | Auth gate via `supabase.auth.getUser()` + ownership query `.eq('id', readingId).eq('therapist_id', user.id).single()` — cross-tenant readingId retorna 404. |
| T-07.1.6-17 | Tampering | JSON body input | mitigate | `typeof readingId !== 'string'` rejected with 400. Sem outros campos lidos do body (orchestrator inputs derivados server-side: therapistId=user.id). |
| T-07.1.6-18 | Repudiation | endpoint POST trail | accept | Anthropic SDK + Supabase já logam requests; canonical_metadata.canonicalized_at registra timestamp na DB. |
| T-07.1.6-19 | Information Disclosure | 500 error message com stack trace | mitigate | console.error inclui stack (server-side log only). Response 500 body inclui `errorMessage` (name + message) — sem stack ao cliente; no PII envolvido. |
| T-07.1.6-20 | Denial of Service | repeated POST hammering | accept | Auth-gated (sem unauth abuse). Founder usage ≤3 readings/dia. Anthropic Tier 1 5 RPM Sonnet limit cobre fail-safe (429 propagates via canonicalizeReading throw → 500). Rate-limit explícito deferred (PROJECT.md envelope). |
| T-07.1.6-21 | Elevation of Privilege | ownership bypass via direct service-role call | mitigate | Endpoint usa `createClient()` (RLS-enforced) para auth gate ANTES de invocar canonicalizeReading (service-role). Sem caminho onde service-role é chamado sem prior auth+ownership check. |
| T-07.1.6-22 | Tampering | canonicalize overwrite via repeated POST (admin Re-canonicalizar abuse) | accept | Idempotency é design intent (D-05). Originais nunca sobrescritos (canonical_storage_path é additivo). Pior caso: caster gasta $0.05 by re-clicking — acceptable cost. |
