    ```typescript
      bbox: IrisBbox
      usage: { input_tokens: number; output_tokens: number }
      cost_usd: number

    /**
     * Single-image bbox fetch. Caller é responsável por:
     *   - downloading the original from Storage (use service client)
     *   - calling Promise.all over 6 images (index.ts orchestrator does this)
     *   - passing the result to isCanonicalAccepted from sanity.ts

    export async function fetchIrisBbox(
      imageBuffer: Buffer,
    ): Promise<FetchBboxResult> {
      // Resize to 1024 long-edge before base64 encoding (probe convention; ~1.8k input tokens)
      const meta = await sharp(imageBuffer).metadata()
      const ow = meta.width
      const oh = meta.height
      if (!ow || !oh) throw new Error('[sonnet-bbox] image metadata missing width/height')
      const scale = Math.max(ow, oh) > VLM_RESIZE_LONG_EDGE ? VLM_RESIZE_LONG_EDGE / Math.max(ow, oh) : 1
      const w = Math.max(1, Math.round(ow * scale))
      const h = Math.max(1, Math.round(oh * scale))
      const resizedBuf = await sharp(imageBuffer).resize(w, h, { fit: 'inside' }).jpeg({ quality: 85 }).toBuffer()
      const base64 = resizedBuf.toString('base64')

      const response = await anthropicClient.messages.create({
        model: SONNET_BBOX_MODEL,
        max_tokens: MAX_TOKENS,
        system: SYSTEM_PROMPT,
        messages: [{
          role: 'user',
          content: [{
            type: 'image',
            source: { type: 'base64', media_type: 'image/jpeg', data: base64 },
          }],
        }],
      })

      const textBlock = response.content.find(b => b.type === 'text')
      if (!textBlock || textBlock.type !== 'text') {
        throw new Error('[sonnet-bbox] no text block in Sonnet response')
      }

      const raw = textBlock.text.trim()
      let parsed: unknown
      try {
        const jsonMatch = raw.match(/\{[\s\S]*\}/)
        parsed = JSON.parse(jsonMatch ? jsonMatch[0] : raw)
      } catch (err) {
        console.error('[sonnet-bbox] parse error:', err, raw.slice(0, 200))
        throw new Error(`[sonnet-bbox] Sonnet returned invalid JSON: ${raw.slice(0, 200)}`)
      }

      const p = parsed as { valid?: unknown; iris_bbox?: { center_x_pct?: unknown; center_y_pct?: unknown; radius_pct?: unknown }; confidence?: unknown }
      const valid = p.valid === true
      const bbox: IrisBbox = {
        center_x_pct: typeof p.iris_bbox?.center_x_pct === 'number' ? p.iris_bbox.center_x_pct : 0.5,
        center_y_pct: typeof p.iris_bbox?.center_y_pct === 'number' ? p.iris_bbox.center_y_pct : 0.5,
        radius_pct:   typeof p.iris_bbox?.radius_pct   === 'number' ? p.iris_bbox.radius_pct   : 0,
        confidence:   typeof p.confidence              === 'number' ? p.confidence              : 0,
        valid,
      }

        input_tokens: response.usage?.input_tokens ?? 0,
        output_tokens: response.usage?.output_tokens ?? 0,
      }
      const cost_usd = estimateCostUsd(usage, PRICING_SONNET_4_6)

      return { bbox, usage, cost_usd }
    }
    ```

    ```typescript
    /**
     * Phase 07.1.6 — canonical crop (sharp).
     *
     * C-02: SEM rotação (VLM rotation morta — provado empiricamente Haiku + Sonnet).
     * C-03: crop floor = max(2.5×r, 0.4×min(W,H)) — mitiga left_lateral radius_pct=0.10.
     *
     * Input: original JPEG buffer (post-EXIF-rotation, server-side download from Storage).
     * Output: 800×800 JPEG buffer (quality 92, lanczos3 kernel).
     */
    import 'server-only'
    import sharp from 'sharp'
    import type { IrisBbox } from '@/lib/anthropic/types'

    const CROP_MULTIPLIER = 2.5
    const CROP_FLOOR_FRACTION = 0.4
    const CROP_OUT_SIZE = 800
    const JPEG_QUALITY = 92

    /**
     * Crop and resize to canonical 800×800.
     *
     * Caller must already have downloaded the original from Storage and called
     * fetchIrisBbox() — bbox is assumed sane (caller invokes isCanonicalAccepted
     * BEFORE calling cropToCanonical, and falls back to original if 'fallback').
     *
     * NÃO faz rotação (C-02). NÃO faz EXIF-rotate (sharp does it implicitly via
     * .rotate() with no args, but here we trust the caller passes a Buffer that's
     * already EXIF-baked — see lib/canonicalize/index.ts downloadImage helper).
     */
    export async function cropToCanonical(
      imageBuffer: Buffer,
      origW: number,
      origH: number,
      bbox: IrisBbox,
    ): Promise<Buffer> {
      const cx = bbox.center_x_pct * origW
      const cy = bbox.center_y_pct * origH
      const r = bbox.radius_pct * Math.min(origW, origH)

      if (r <= 1) {
        throw new Error(`[canonicalize/crop] radius_pct ${bbox.radius_pct} → ${r}px too small to crop`)
      }

      // C-03: floor = max(2.5×r, 0.4×min(W,H))
      const rawCrop = Math.round(CROP_MULTIPLIER * r)
      const floorCrop = Math.round(CROP_FLOOR_FRACTION * Math.min(origW, origH))
      let cropSide = Math.max(rawCrop, floorCrop)
      // Clamp to image bounds (cropSide can't exceed image)
      cropSide = Math.min(cropSide, origW, origH)

      let left = Math.round(cx - cropSide / 2)
      let top = Math.round(cy - cropSide / 2)
      // Clamp to keep extract() within image bounds
      left = Math.max(0, Math.min(left, origW - cropSide))
      top = Math.max(0, Math.min(top, origH - cropSide))

      return await sharp(imageBuffer)
        .extract({ left, top, width: cropSide, height: cropSide })
        .resize(CROP_OUT_SIZE, CROP_OUT_SIZE, { fit: 'cover', kernel: 'lanczos3' })
        .jpeg({ quality: JPEG_QUALITY })
        .toBuffer()
    }
    ```

    ```typescript
     * Phase 07.1.6 — canonicalize orchestrator.
     *
     * Called by:
     *   - app/api/capture/canonicalize/route.ts (Wave 1 Plan 04) — POST endpoint, hit by:
     *       - finalizeReadingAction (Wave 2 Plan 05) after 6/6 uploads
     *       - /admin/calibration "Re-canonicalizar" button (Wave 2 Plan 06)
     *
     * Flow:
     *   1. D-04 env-flag check (CANONICAL_CAPTURE_ENABLED !== 'false'; default ON)
     *   2. Fetch reading_images (service-role, bypass RLS — caller already auth-checked)
     *   3. 6× Sonnet bbox in parallel via Promise.all
     *   4. Group bboxes by eye (3 angles per eye)
     *   5. Per-image: isCanonicalAccepted(bbox, peers) → 'ok' | 'fallback'
     *   6. For 'ok' images: download original → cropToCanonical → upload to canonical/ path → upsert canonical_storage_path
     *   7. Aggregate usage + costs → return per-image result + summary
     *
     * Idempotent: upsert onConflict 'reading_id,eye,angle' allows re-canonicalize
     * (admin Re-canonicalizar button calls this same path; D-05).
     */

    import { createServiceClient } from '@/lib/supabase/service'
    import { fetchIrisBbox } from './sonnet-bbox'
    import { cropToCanonical } from './crop'
    import { isCanonicalAccepted } from './sanity'
    import { buildCanonicalStoragePath } from './storage-path'
    import type { IrisBbox, CanonicalStatus, CanonicalMetadata } from '@/lib/anthropic/types'
    import type { Eye } from '@/lib/capture/iris-geometry'
    import type { Angle } from '@/lib/capture/sequence'
    import sharp from 'sharp'

    const BUCKET = 'iris-captures'

    export interface CanonicalizeResult {
      eye: Eye
      angle: Angle
      canonical_storage_path: string | null
      canonical_status: CanonicalStatus
      cost_usd: number
      bbox: IrisBbox | null   // null on disabled or download/fetch error
      error?: string

      results: CanonicalizeResult[]
      metadata: CanonicalMetadata
    }

    /** Internal — single image bbox fetch with error capture. */

      eye: Eye
      angle: Angle
      storage_path: string
      bbox: IrisBbox | null
      bufferBaked: Buffer | null         // EXIF-baked original buffer (reused for crop)
      origW: number
      origH: number
      input_tokens: number
      output_tokens: number
      cost_usd: number

    /**
     * Orchestrate canonicalize for 1 reading.
     *
     * @param readingId - target reading
     * @param therapistId - owner (passed by caller; route.ts derives from session)
     */
    export async function canonicalizeReading(
      readingId: string,
      therapistId: string,
    ): Promise<CanonicalizeReadingOutput> {
      const enabled = process.env.CANONICAL_CAPTURE_ENABLED !== 'false'  // D-04 default ON
      const service = createServiceClient()

      // Fetch the 6 reading_images
      const { data: images, error: imgError } = await service
        .from('reading_images')
        .select('eye, angle, storage_path')
        .eq('reading_id', readingId)

      if (imgError) {
        throw new Error(`[canonicalize] fetch images failed: ${imgError.message}`)
      }
      if (!images || images.length === 0) {
        throw new Error(`[canonicalize] no images found for reading ${readingId}`)
      }

      // D-04 kill-switch: skip Sonnet calls entirely, return all-disabled
      if (!enabled) {
        const disabledResults: CanonicalizeResult[] = images.map(img => ({
          eye: img.eye as Eye,
          angle: img.angle as Angle,
          canonical_storage_path: null,
          canonical_status: 'disabled',
          cost_usd: 0,
          bbox: null,
        }))
        const metadata: CanonicalMetadata = {
          sonnet_input_tokens: 0,
          sonnet_output_tokens: 0,
          cost_usd: 0,
          status_summary: { ok: 0, fallback: 0, disabled: disabledResults.length },
          canonicalized_at: new Date().toISOString(),
        }
        return { results: disabledResults, metadata }
      }

      // Step 1: 6 Sonnet calls in parallel (~3-5s wall clock; ~$0.05 total)
      const bboxResults: PerImageBboxResult[] = await Promise.all(
        images.map(async (img): Promise<PerImageBboxResult> => {
          const base: PerImageBboxResult = {
            eye: img.eye as Eye,
            angle: img.angle as Angle,
            storage_path: img.storage_path,
            bbox: null,
            bufferBaked: null,
            origW: 0,
            origH: 0,
            input_tokens: 0,
            output_tokens: 0,
            cost_usd: 0,
          }
          try {
            // Download from Storage; bake EXIF (same as probe-haiku-iris-landmarks.mjs L161-170)
            const { data: blob, error: dlError } = await service.storage.from(BUCKET).download(img.storage_path)
            if (dlError || !blob) {
              throw new Error(`storage download failed: ${dlError?.message ?? 'unknown'}`)
            }
            const ab = await blob.arrayBuffer()
            const rawBuf = Buffer.from(ab)
            const baked = await sharp(rawBuf).rotate().toBuffer()
            const meta = await sharp(baked).metadata()
            base.bufferBaked = baked
            base.origW = meta.width ?? 0
            base.origH = meta.height ?? 0

            const { bbox, usage, cost_usd } = await fetchIrisBbox(baked)
            base.bbox = bbox
            base.input_tokens = usage.input_tokens
            base.output_tokens = usage.output_tokens
            base.cost_usd = cost_usd
            return base
          } catch (err) {
            const msg = err instanceof Error ? err.message : 'unknown error'
            console.error(`[canonicalize] bbox fetch failed eye=${img.eye} angle=${img.angle}: ${msg}`)
            base.error = msg
            return base
          }
        })
      )

      // Step 2: group bboxes per eye for cross-angle peer set
      const bboxesByEye: Record<string, IrisBbox[]> = {}
      for (const r of bboxResults) {
        if (r.bbox) {
          if (!bboxesByEye[r.eye]) bboxesByEye[r.eye] = []
          bboxesByEye[r.eye].push(r.bbox)
        }
      }

      // Step 3: per-image trust gate + (if 'ok') crop + upload + upsert in parallel
      const results: CanonicalizeResult[] = await Promise.all(
        bboxResults.map(async (r): Promise<CanonicalizeResult> => {
          // Gate
          if (!r.bbox || r.error || !r.bufferBaked) {
            return {
              eye: r.eye,
              angle: r.angle,
              canonical_storage_path: null,
              canonical_status: 'fallback',
              cost_usd: r.cost_usd,
              bbox: r.bbox,
              error: r.error,
            }
          }
          const peers = (bboxesByEye[r.eye] ?? []).filter(b => b !== r.bbox)
          const status = isCanonicalAccepted(r.bbox, peers)
          if (status !== 'ok') {
            return {
              eye: r.eye,
              angle: r.angle,
              canonical_storage_path: null,
              canonical_status: 'fallback',
              cost_usd: r.cost_usd,
              bbox: r.bbox,
            }
          }

          // Crop + upload
          try {
            const canonicalBuf = await cropToCanonical(r.bufferBaked, r.origW, r.origH, r.bbox)
            const path = buildCanonicalStoragePath(therapistId, readingId, r.eye, r.angle)
            const { error: upError } = await service.storage.from(BUCKET).upload(path, canonicalBuf, {
              contentType: 'image/jpeg',
              upsert: true,
            })
            if (upError) throw new Error(`storage upload failed: ${upError.message}`)

            // Persist canonical_storage_path. Match upsert pattern from lib/capture/upload.ts L81-98.
            const { error: dbError } = await service
              .from('reading_images')
              .update({ canonical_storage_path: path })
              .eq('reading_id', readingId)
              .eq('eye', r.eye)
              .eq('angle', r.angle)
            if (dbError) throw new Error(`db update failed: ${dbError.message}`)

            return {
              eye: r.eye,
              angle: r.angle,
              canonical_storage_path: path,
              canonical_status: 'ok',
              cost_usd: r.cost_usd,
              bbox: r.bbox,
            }
          } catch (err) {
            const msg = err instanceof Error ? err.message : 'unknown error'
            console.error(`[canonicalize] crop/upload failed eye=${r.eye} angle=${r.angle}: ${msg}`)
            return {
              eye: r.eye,
              angle: r.angle,
              canonical_storage_path: null,
              canonical_status: 'fallback',
              cost_usd: r.cost_usd,
              bbox: r.bbox,
              error: msg,
            }
          }
        })
      )

      // Aggregate metadata
      const totalInput = bboxResults.reduce((sum, r) => sum + r.input_tokens, 0)
      const totalOutput = bboxResults.reduce((sum, r) => sum + r.output_tokens, 0)
      const totalCost = bboxResults.reduce((sum, r) => sum + r.cost_usd, 0)
      const summary: Record<CanonicalStatus, number> = { ok: 0, fallback: 0, disabled: 0 }
      for (const r of results) summary[r.canonical_status]++
      const metadata: CanonicalMetadata = {
        sonnet_input_tokens: totalInput,
        sonnet_output_tokens: totalOutput,
        cost_usd: totalCost,
        status_summary: summary,
        canonicalized_at: new Date().toISOString(),
      }

      // Persist aggregate metadata to readings.canonical_metadata
      const { error: metaError } = await service
        .from('readings')
        .update({ canonical_metadata: metadata })
        .eq('id', readingId)
      if (metaError) {
        console.error(`[canonicalize] update canonical_metadata failed: ${metaError.message}`)
        // Non-fatal: per-image results já persistidos. Caller pode log.
      }

      return { results, metadata }
    }
    ```
