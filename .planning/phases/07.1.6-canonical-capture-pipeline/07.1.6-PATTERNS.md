```typescript
export function buildCanonicalStoragePath(
  therapistId: string,
  readingId: string,
  eye: Eye,
  angle: Angle,
): string {
  validateSegment(therapistId, 'therapistId')
  validateSegment(readingId, 'readingId')
  return `${therapistId}/${readingId}/canonical/${eye}_${angle}.jpg`
}

function validateSegment(value: string, name: string): void {
  if (!value || typeof value !== 'string') {
    throw new Error(`[storage-path] ${name} é obrigatório`)
  }
  if (value.includes('/') || value.includes('..') || value.includes('\\')) {
    throw new Error(`[storage-path] ${name} contém caracteres inválidos`)
  }
}
```

```typescript
// No imports — pure math only. No 'server-only' needed.

export interface IrisBbox {
  center_x_pct: number
  center_y_pct: number
  radius_pct: number
  confidence: number
}

export type CanonicalStatus = 'ok' | 'fallback' | 'disabled'

// Threshold constants (empirically derived — CONTEXT.md <specifics>)
const GEOM_CENTER_MIN = 0.20
const GEOM_CENTER_MAX = 0.80
const GEOM_RADIUS_MIN = 0.05
const GEOM_RADIUS_MAX = 0.30
const CROSS_ANGLE_OUTLIER_THRESHOLD = 0.08  // D-02

export function isGeometricallySane(bbox: IrisBbox): boolean {
  return (
    bbox.center_x_pct >= GEOM_CENTER_MIN &&
    bbox.center_x_pct <= GEOM_CENTER_MAX &&
    bbox.center_y_pct >= GEOM_CENTER_MIN &&
    bbox.center_y_pct <= GEOM_CENTER_MAX &&
    bbox.radius_pct >= GEOM_RADIUS_MIN &&
    bbox.radius_pct <= GEOM_RADIUS_MAX
  )
}

/** Returns true if this bbox is an outlier vs peer bboxes from the same eye. */
export function isCrossAngleOutlier(bbox: IrisBbox, peers: IrisBbox[]): boolean {
  if (peers.length < 2) return false
  const medianX = median(peers.map(p => p.center_x_pct))
  const medianY = median(peers.map(p => p.center_y_pct))
  return (
    Math.abs(bbox.center_x_pct - medianX) > CROSS_ANGLE_OUTLIER_THRESHOLD ||
    Math.abs(bbox.center_y_pct - medianY) > CROSS_ANGLE_OUTLIER_THRESHOLD
  )
}

export function isCanonicalAccepted(bbox: IrisBbox, peers: IrisBbox[]): CanonicalStatus {
  if (!isGeometricallySane(bbox)) return 'fallback'
  if (isCrossAngleOutlier(bbox, peers)) return 'fallback'
  return 'ok'
}

function median(values: number[]): number {
  const sorted = [...values].sort((a, b) => a - b)
  const mid = Math.floor(sorted.length / 2)
  return sorted.length % 2 === 0
    ? (sorted[mid - 1] + sorted[mid]) / 2
    : sorted[mid]
}
```

### `apps/web/lib/canonicalize/crop.ts` (pure-lib, transform)

**Analog:** `apps/web/scripts/probe-haiku-iris-landmarks.mjs` — `applyCrop()` function (lines 213-271)

**Core crop pattern** (extract from probe script, adapt C-03 floor):
```typescript
import sharp from 'sharp'

const CROP_MULTIPLIER = 2.5
const CROP_OUT_SIZE = 800

export async function cropToCanonical(
  buf: Buffer,
  origW: number,
  origH: number,
  bbox: IrisBbox,
): Promise<Buffer> {
  const cx = bbox.center_x_pct * origW
  const cy = bbox.center_y_pct * origH
  const r = bbox.radius_pct * Math.min(origW, origH)

  if (r <= 1) throw new Error(`radius_pct ${bbox.radius_pct} → ${r}px too small`)

  // C-03: floor = max(2.5×r, 0.4×min(W,H))
  const rawCrop = Math.round(CROP_MULTIPLIER * r)
  const floorCrop = Math.round(0.4 * Math.min(origW, origH))
  let cropSide = Math.max(rawCrop, floorCrop)
  cropSide = Math.min(cropSide, origW, origH)

  let left = Math.round(cx - cropSide / 2)
  let top = Math.round(cy - cropSide / 2)
  left = Math.max(0, Math.min(left, origW - cropSide))
  top = Math.max(0, Math.min(top, origH - cropSide))

  return await sharp(buf)
    .extract({ left, top, width: cropSide, height: cropSide })
    .resize(CROP_OUT_SIZE, CROP_OUT_SIZE, { fit: 'cover', kernel: 'lanczos3' })
    .jpeg({ quality: 92 })
    .toBuffer()
}
```

Key differences vs probe script:
- NO rotation (C-02: terapeuta orienta; probe had `rotatePointAroundCenter` + `sharp.rotate()` — omit entirely)
- C-03 floor `max(2.5×r, 0.4×min(W,H))` replaces probe's bare `CROP_MULTIPLIER * r`
- Output as Buffer (not file); caller handles Storage upload

```typescript
export interface CanonicalizeResult {
  eye: Eye
  angle: Angle
  canonical_storage_path: string | null   // null = fallback
  canonical_status: CanonicalStatus
  cost_usd: number
}

export async function canonicalizeReading(
  readingId: string,
  therapistId: string,
): Promise<CanonicalizeResult[]> {
  const enabled = process.env.CANONICAL_CAPTURE_ENABLED !== 'false'
  if (!enabled) {
    // return all-disabled without calling Sonnet
  }

  const service = createServiceClient()
  const { data: images } = await service
    .from('reading_images')
    .select('eye, angle, storage_path')
    .eq('reading_id', readingId)

  // Step 1: fetch all bboxes in parallel (Promise.all — wall-clock ~3-5s)
  const bboxResults = await Promise.all(images.map(img => fetchBboxForImage(img)))

  // Step 2: group by eye for cross-angle consistency check (D-02)
  const byEye: Record<string, IrisBbox[]> = {}
  for (const r of bboxResults) {
    if (r.bbox) (byEye[r.eye] ??= []).push(r.bbox)
  }

  // Step 3: per-image crop + upload in parallel
  return Promise.all(bboxResults.map(async r => {
    const peers = (byEye[r.eye] ?? []).filter(b => b !== r.bbox)
    const status = r.bbox ? isCanonicalAccepted(r.bbox, peers) : 'fallback'
    if (status !== 'ok' || !r.bbox) {
      return { eye: r.eye, angle: r.angle, canonical_storage_path: null, canonical_status: 'fallback', cost_usd: r.cost_usd }
    }
    const canonicalPath = buildCanonicalStoragePath(therapistId, readingId, r.eye, r.angle)
    // crop → upload (uploadBuffer pattern from upload.ts)
    // update reading_images.canonical_storage_path
    return { eye: r.eye, angle: r.angle, canonical_storage_path: canonicalPath, canonical_status: 'ok', cost_usd: r.cost_usd }
  }))
}
```
