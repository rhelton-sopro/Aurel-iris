```tsx
export interface AnaliseClientProps {
  readingId: string
  hasInitialReport: boolean
  // ... existing
  reportVersion?: '1.0' | '2.0'  // NEW — passed from page.tsx
}

// V1 simplest path: client counts top-level key occurrences in the running
// buffer using the same next-key-start regex from Plan 07.4-03 stream-parser-v2,
// inlined client-side (stream-parser-v2 imports 'server-only').
const V2_KEYS_ORDERED = [
  'executive_summary', 'constitutional_pattern', 'systems_with_tendency',
  'integrative_axes', 'bilateral_findings', 'therapeutic_synthesis',
  'priority_focus', 'clinical_note',
] as const

function detectBlocksReceived(buffer: string): number {
  // Edge case: when the stream completes successfully, the final `clinical_note` key
  // is detected complete by spotting `,"advanced_analysis":`. Truncated streams that
  // never reach advanced_analysis will leave clinical_note Skeleton on the client UI
  // even when zod parse succeeds on the final buffer. This is a cosmetic limitation
  // (the final report renders correctly; only the streaming progress indicator stays
  // on the last block). Accepted as a documented V1 limitation.
  let count = 0
  for (let i = 0; i < V2_KEYS_ORDERED.length - 1; i++) {
    const next = V2_KEYS_ORDERED[i + 1]
    const re = new RegExp(`,\\s*"${next}"\\s*:`)
    if (re.test(buffer)) count++
    else break
  }
  // Also try to detect clinical_note completion via advanced_analysis sentinel
  if (count === V2_KEYS_ORDERED.length - 1) {
    const finalRe = /,\s*"advanced_analysis"\s*:/
    if (finalRe.test(buffer)) count++
  }
  return count
}

// In the streaming JSX, switch:
{reportVersion === '2.0' ? (
  <AdaptiveAnalysisStream blocksReceived={detectBlocksReceived(streamBuffer)} error={err} />
) : (
  <AnalysisStream sectionsReceived={sectionsReceivedFromLegacyParser} error={err} />
)}
```
