```js
#!/usr/bin/env node
/**
 * audit-vocabulary.mjs — CI gate scanning apps/web source files for forbidden vocabulary.
 *
 * Phase 7.4 extension (D-VOC1, D-VOC2):
 *   3 pattern sets sourced from lib/anthropic/forbidden-terms.json:
 *     - iridological_jargon (D-VOC1 set 1) — proibido no relatório padrão
 *     - sopro_vocab (D-VOC1 set 2) — proibido absoluto em Iris Codex copy
 *     - lgpd (D-VOC1 set 3, existing) — diagnóstico|tratamento|cura
 *
 * EXCLUDE_SUBPATHS preserved from existing pattern + forbidden-terms.json
 * (term source — scanning would self-match).
 *
 * ALLOWLIST_MARKER preserved — files with the marker are SKIPPED entirely
 * (e.g., system.md instructs LLM to AVOID forbidden vocab; audit.ts implements
 * the runtime detector and quotes terms in docs comments).
 *
 * Phase 7.4 | Plan 07.4-02 | Decisões: D-VOC1, D-VOC2
 */
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { join, dirname } from 'node:path'
import { fileURLToPath } from 'node:url'

const __dirname = dirname(fileURLToPath(import.meta.url))
const ROOT = join(__dirname, '..')

// Load term source
const termsPath = join(ROOT, 'lib/anthropic/forbidden-terms.json')
const terms = JSON.parse(readFileSync(termsPath, 'utf8'))

// Build categorized regex sets. Escape regex metachars in each term.
function escapeRegex(str) {
  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}

function buildSet(termList) {
  // Word-boundary regex for each term — \b...\b with unicode flag for accented chars.
  // Combined alternation; case-insensitive.
  const escaped = termList.map(escapeRegex).join('|')
  return new RegExp(`\\b(${escaped})\\b`, 'iu')
}

const PATTERNS = {
  iridological_jargon: buildSet(terms.iridological_jargon),
  sopro_vocab: buildSet(terms.sopro_vocab),
  lgpd: buildSet(terms.lgpd),
}

const EXTENSIONS = new Set(['.ts', '.tsx', '.js', '.jsx', '.mjs', '.cjs', '.md'])
const DIRS = ['app', 'components', 'lib/rag', 'lib/anthropic', 'prompts']
const EXCLUDE_SUBPATHS = [
  '/app/admin/',
  '/components/calibration/',
  '/app/actions/calibration.test.',
  '/lib/anthropic/forbidden-terms.json', // term source — self-match
  '/lib/anthropic/__tests__/fixtures/',   // synthetic test data may include forbidden vocab
]
const ALLOWLIST_MARKER = 'audit-vocabulary:allowlist'

function collectFiles(dir) {
  const out = []
  let entries
  try { entries = readdirSync(dir) } catch { return out }
  for (const entry of entries) {
    const full = join(dir, entry)
    let s
    try { s = statSync(full) } catch { continue }
    if (s.isDirectory()) {
      if (entry === 'node_modules' || entry === '.next' || entry === 'dist') continue
      out.push(...collectFiles(full))
    } else if (s.isFile()) {
      const ext = full.slice(full.lastIndexOf('.'))
      if (EXTENSIONS.has(ext)) out.push(full)
    }
  }
  return out
}

const matches = []  // { category, file, line, text }

for (const dir of DIRS) {
  const absDir = join(ROOT, dir)
  for (const file of collectFiles(absDir)) {
    const normalized = file.replace(/\\/g, '/')
    if (EXCLUDE_SUBPATHS.some(prefix => normalized.includes(prefix))) continue

    const content = readFileSync(file, 'utf8')
    if (content.includes(ALLOWLIST_MARKER)) continue

    const lines = content.split('\n')
    for (let i = 0; i < lines.length; i++) {
      for (const [category, regex] of Object.entries(PATTERNS)) {
        if (regex.test(lines[i])) {
          matches.push({
            category,
            file: normalized,
            line: i + 1,
            text: lines[i].trim().slice(0, 200),
          })
        }
      }
    }
  }
}

if (matches.length > 0) {
  console.error(`✗ audit-vocabulary: ${matches.length} forbidden term hit(s) found.\n`)
  // Group by category
  const grouped = matches.reduce((acc, m) => {
    (acc[m.category] = acc[m.category] || []).push(m)
    return acc
  }, {})
  for (const [category, hits] of Object.entries(grouped)) {
    console.error(`[${category}] (${hits.length} hits):`)
    for (const h of hits) console.error(`  ${h.file}:${h.line}: ${h.text}`)
    console.error('')
  }
  process.exit(1)
}

console.log(`✓ audit-vocabulary: clean across ${DIRS.length} dirs, ${Object.keys(PATTERNS).length} pattern sets.`)
process.exit(0)
```

# Iris Codex — Analista clínico-funcional

Você é o **analista clínico-funcional do Iris Codex**, uma plataforma de relatórios iridológicos funcionais adaptativos para terapeutas integrativos brasileiros. Sua função é gerar um **relatório clínico-funcional adaptativo** em formato JSON estruturado a partir de:

1. Contexto do cliente (nome, idade, sexo, queixa principal)
2. Tendências sistêmicas pré-mapeadas (output da engine de mapeamento Iris Codex que traduz sinais iridológicos brutos em tendências funcionais)
3. Trechos de conhecimento iridológico (RAG — ancorados nas escolas relevantes às tendências)

Você **NÃO** recebe diretamente os sinais iridológicos brutos. A engine de mapeamento já fez essa tradução. Sua tarefa é redigir um relatório **clínico-funcional** que o terapeuta integrativo (não iridologista) consegue ler e usar na anamnese.

---

## Identidade de marca (D-BR3)

**Iris Codex** é uma plataforma científico-clínica. Linguagem **funcional e clínica**, não espiritual. O Iris Codex é separado e distinto da linha **Sopro da Origem** (linha espiritual da fundadora, fora deste produto). Você **nunca** usa vocabulário espiritual de Sopro da Origem no relatório Iris Codex.

## 7 Princípios de operação (LEIA TODOS antes de gerar)

### Princípio 1 — Você não diagnostica

Você **NUNCA** usa as palavras: `diagnóstico`, `tratamento`, `cura`. LGPD-06 é linha vermelha permanente. Sempre fraseie como:

- ✓ "tendência a sobrecarga hepática"
- ✓ "sugere considerar avaliação de..."
- ✓ "considere investigar..."
- ✗ "diagnóstico de fígado sobrecarregado"
- ✗ "tratamento indicado: detox"

### Princípio 6 — Regra das duas vozes

Mantenha separação entre dois registros, sem misturar na mesma cláusula:

1. **Voz factual firme** — descrição de manifestações clínicas documentadas ("o cliente relata sinusite recorrente"). Use indicativo presente direto.
2. **Voz hipotética** — interpretações e tendências inferidas ("sugere considerar investigação de drenagem comprometida"). Use "sugere", "considere", "pode indicar", "tende a".

**Errado:** "O fígado sobrecarregado está causando intolerância alimentar."
